"""Root-only measured TV office composition; retain access and both wall artworks."""
import argparse, copy, json, msvcrt
import nbtlib
import numpy as np
import regional_voxels as vox
from r52_spatial_plan import BASE
from survey_r52_original_complex import measure
from query_blocks import iter_block_entities
from prepare_continuous_upright_civil_r52 import cuboids
from extend_world_depth_r52 import acquire_world_lock


def packed(x,y,z):
    n=((x&0x3ffffff)<<38)|((z&0x3ffffff)<<12)|(y&4095)
    return n-(1<<64) if n>=1<<63 else n


def artwork(pos,up):
    x,y,z=pos;back=(x,y-1 if up else y+1,z)
    return nbtlib.Compound(dict(id=nbtlib.String('projectseele:wall_artwork'),x=nbtlib.Int(x),y=nbtlib.Int(y),z=nbtlib.Int(z),
        Artwork=nbtlib.String('tree_ink'),SurfaceFacing=nbtlib.String('up' if up else 'down'),Facing=nbtlib.Int(0),
        Width=nbtlib.Float(24),Height=nbtlib.Float(24),OffsetX=nbtlib.Double(.5),OffsetZ=nbtlib.Double(.5),
        OffsetY=nbtlib.Double(.015 if up else .985),Backing=nbtlib.Long(packed(*back)),
        Tint=nbtlib.Int(int('ff996080',16)-(1<<32))))


def main():
    ap=argparse.ArgumentParser();ap.add_argument('--apply',action='store_true');args=ap.parse_args()
    out=BASE/'final_map_audit/commander_original_art';stage='tv_office_clear_axis_and_horizontal_art'
    assert not list((out/stage).glob('applied_*')), 'Do not replay'
    source=np.load(out/'CURRENT.npz');lo=tuple(map(int,source['origin']));before=source['states'];names=list(map(str,source['palette']))
    for s in ('minecraft:black_concrete','projectseele:wall_artwork','minecraft:air'):
        if s not in names:names.append(s)
    ids={s:i for i,s in enumerate(names)};after=before.copy()
    hi=(lo[0]+before.shape[2]-1,lo[1]+before.shape[0]-1,lo[2]+before.shape[1]-1)
    tags=dict(iter_block_entities(BASE/'WORLD_R52','projectseele:geofront',lo,hi))
    def ix(p):x,y,z=p;return y-lo[1],z-lo[2],x-lo[0]
    retired=[]
    for y in (-422,-421,-420):
        for z in range(320,340):
            for x in range(24,37):
                p=x,y,z;s=names[int(before[ix(p)])]
                furniture=(s.startswith(('another_furniture:black_stool[','another_furniture:red_stool[',
                    'projectseele:command_seat_back[','minecraft:dark_oak_wall_sign[','minecraft:mangrove_wall_sign[',
                    'minecraft:black_wall_banner[','minecraft:yellow_wall_banner['))
                    or (28<=x<=32 and 322<=z<=336 and s=='minecraft:polished_blackstone_slab[type=top,waterlogged=false]'))
                if not furniture:continue
                tag=tags.get(p)
                if tag is not None:
                    if str(tag['id'])=='minecraft:sign':
                        for side in ('front_text','back_text'):
                            for m in tag.get(side,{}).get('messages',[]):
                                q=json.loads(str(m));assert q=={'text':''},('Nonempty furniture arm',p,q)
                    else:assert str(tag['id'])=='minecraft:banner' and not tag.get('Patterns',[]),(p,tag)
                after[ix(p)]=ids['minecraft:air'];retired.append(dict(pos=p,state=s))
    assert sum(r['state'].startswith('minecraft:polished_blackstone_slab') for r in retired)==75
    # Dark unbroken main floor; the occupied stairs, outer stepped shell and
    # secured archive behind the original tree wall are preserved.
    floor_count=0
    for x in range(6,55):
        for z in range(304,339):
            p=x,-423,z;s=names[int(before[ix(p)])]
            if s in ('minecraft:white_concrete','minecraft:red_concrete','minecraft:black_concrete','minecraft:polished_blackstone'):
                after[ix(p)]=ids['minecraft:black_concrete'];floor_count+=1
    # A real ceiling sits inside the measured stepped envelope, 14m above
    # the floor. No outside shell/glass or the crown is modified.
    for x in range(16,45):
        for z in range(313,342):
            p=x,-408,z
            assert names[int(before[ix(p)])] in ('minecraft:air','minecraft:void_air'),('Ceiling touches an original component',p,names[int(before[ix(p)])])
            after[ix(p)]=ids['minecraft:black_concrete']
    floor_pos=(30,-422,327);ceiling_pos=(30,-409,327)
    for p in (floor_pos,ceiling_pos):
        assert names[int(after[ix(p)])]=='minecraft:air'
        after[ix(p)]=ids['projectseele:wall_artwork']
    # Refinish the existing tree plate backing, but do not remove its hidden
    # archive enclosure, clearance3 card readers or door.
    for y in range(-422,-408):
        for x in range(20,41):
            p=x,y,340
            if names[int(before[ix(p)])] in ('minecraft:white_concrete','minecraft:smooth_quartz'):
                after[ix(p)]=ids['minecraft:black_concrete']
    for p,t in tags.items():
        if before[ix(p)]!=after[ix(p)]:assert any(tuple(r['pos'])==p for r in retired),('Protected device changed',p,str(t['id']))
    vox.WORLD=BASE/'WORLD_R52';vox.OUT=out;vox.DIM='projectseele:geofront';paint=vox.Painter(vertical_bounds=(-960,320))
    n=cuboids(paint,lo,before,after,names,'R52_TV_commander_office_composition')
    paint.block_entities[floor_pos]=artwork(floor_pos,True);paint.block_entities[ceiling_pos]=artwork(ceiling_pos,False)
    tree=(30,-415,339);old=tags[tree];new=copy.deepcopy(old)
    new['Artwork']=nbtlib.String('tree_ink');new['Tint']=nbtlib.Int(int('ff996080',16)-(1<<32))
    paint.update_block_entity(tree,'projectseele:wall_artwork',old,new,'R52_existing_tree_ink_no_paper_backdrop')
    paint.meta.update(root_author=True,original_nerv_and_tree_wall_anchors_preserved=True,card_readers_archive_doors_preserved=True,
        stepped_shell_and_all_glass_preserved=True,reference='TV Gendo office: sparse axis, single desk, ceiling and floor tree',
        requires_surface_facing_artwork_code=True,native_art_review_pending=True)
    target=paint.save_plan(stage)
    report=dict(world_written=False,changed_cells=n,retired_conference_furniture=retired,floor_cells=floor_count,
        new_art_anchors=[floor_pos,ceiling_pos],existing_tree_anchor=tree,ceiling_cells=29*29,
        original_desk_and_deputy_seats_preserved=True,original_devices=len(tags),native_art_accepted=False)
    np.savez_compressed(target/'EXPECTED.npz',origin=lo,states=after,palette=names)
    if args.apply:
        current,pal=measure(lo,hi);assert np.array_equal(np.array(names)[before],np.array(pal)[current])
        lock=acquire_world_lock()
        try:report.update(world_written=True,receipt=paint.apply(stage,session_lock=lock))
        finally:msvcrt.locking(lock.fileno(),msvcrt.LK_UNLCK,1);lock.close()
    (target/'PLAN.json').write_text(json.dumps(report,indent=2),'utf8')
    print(json.dumps({k:v for k,v in report.items()if k!='retired_conference_furniture'}),flush=True)


if __name__=='__main__':main()
