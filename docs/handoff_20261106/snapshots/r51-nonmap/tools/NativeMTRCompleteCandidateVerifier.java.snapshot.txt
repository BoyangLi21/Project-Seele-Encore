import org.mtr.core.data.*;
import org.mtr.core.serializer.*;
import org.mtr.core.tool.Utilities;
import org.mtr.libraries.com.google.gson.*;
import java.nio.file.*;
import java.lang.reflect.*;
import java.util.*;
import java.util.zip.*;
import java.security.MessageDigest;
import org.mtr.libraries.org.msgpack.core.MessagePack;

/** Executes only isolated MTR core objects; no world, simulator, or DB writer. */
public final class NativeMTRCompleteCandidateVerifier
{
    private static JsonObject serialize(SerializedDataBase value)
    {JsonObject q=new JsonObject();value.serializeData(new JsonWriter(q));return q;}
    private static Object field(Object o,String name)throws Exception
    {Field f=o.getClass().getDeclaredField(name);f.setAccessible(true);return f.get(o);}
    private static String key(JsonObject a,JsonObject b)
    {
        String x=a.toString(),y=b.toString();return x.compareTo(y)<0?x+"|"+y:y+"|"+x;
    }
    private static Data load(JsonObject records,boolean vehicles)throws Exception
    {
        Data data=new Data(){};
        for(var kind:records.entrySet())for(var value:kind.getValue().getAsJsonArray())
        {
            ReaderBase r=new JsonReader(value.getAsJsonObject().get("data"));
            switch(kind.getKey())
            {
                case "stations" -> data.stations.add(new Station(r,data));
                case "platforms" -> data.platforms.add(new Platform(r,data));
                case "rails" -> data.rails.add(new Rail(r));
                case "routes" -> data.routes.add(new Route(r,data));
                case "depots" -> data.depots.add(new Depot(r,data));
                case "sidings" -> data.sidings.add(new Siding(r,data));
                default -> {}
            }
        }
        data.sync();
        if(vehicles)
        {
            for(Depot d:data.depots)d.writePathCache();
            Method m=Siding.class.getDeclaredMethod("writePathCache");m.setAccessible(true);
            for(Siding s:data.sidings)m.invoke(s);
            for(Siding s:data.sidings)s.init();
        }
        return data;
    }
    private static String sha(byte[] bytes)throws Exception
    {return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(bytes));}
    private static Data loadBinaryPatch(Path manifestPath)throws Exception
    {
        JsonObject manifest=JsonParser.parseString(Files.readString(manifestPath)).getAsJsonObject();
        Map<String,byte[]> members=new LinkedHashMap<>();String marker="mtr/projectseele/geofront/";
        try(ZipFile zip=new ZipFile(manifest.get("archive").getAsString()))
        {
            var entries=zip.entries();while(entries.hasMoreElements())
            {
                var entry=entries.nextElement();int offset=entry.getName().indexOf(marker);
                if(offset<0||entry.isDirectory())continue;
                members.put(entry.getName().substring(offset),zip.getInputStream(entry).readAllBytes());
            }
        }
        for(var value:manifest.getAsJsonArray("protected_original_members"))
        {
            JsonObject row=value.getAsJsonObject();String name=row.get("member").getAsString();name=name.substring(name.indexOf(marker));
            if(!sha(members.get(name)).equals(row.get("sha256").getAsString()))throw new AssertionError("Protected original bytes changed "+name);
        }
        for(var value:manifest.getAsJsonArray("operations"))
        {
            JsonObject op=value.getAsJsonObject();String name=op.get("target_relative_path").getAsString(),action=op.get("action").getAsString();
            if(!action.equals("create")&&!sha(members.get(name)).equals(op.get("before_sha256").getAsString()))throw new AssertionError("Patch before SHA "+name);
            if(action.equals("delete")){members.remove(name);continue;}
            byte[] after=Files.readAllBytes(manifestPath.getParent().resolve("after").resolve(name));
            if(!sha(after).equals(op.get("after_sha256").getAsString()))throw new AssertionError("Patch after SHA "+name);
            if(action.equals("create")&&members.containsKey(name))throw new AssertionError("New rail path collision "+name);
            members.put(name,after);
        }
        Data data=new Data(){};
        for(var entry:members.entrySet())try(var unpacker=MessagePack.newDefaultUnpacker(entry.getValue()))
        {
            ReaderBase r=new MessagePackReader(unpacker);String type=entry.getKey().substring(marker.length()).split("/")[0];
            switch(type)
            {
                case "stations" -> data.stations.add(new Station(r,data));
                case "platforms" -> data.platforms.add(new Platform(r,data));
                case "rails" -> data.rails.add(new Rail(r));
                case "routes" -> data.routes.add(new Route(r,data));
                case "depots" -> data.depots.add(new Depot(r,data));
                case "sidings" -> data.sidings.add(new Siding(r,data));
                default -> {}
            }
        }
        data.sync();for(Depot d:data.depots)d.writePathCache();
        Method m=Siding.class.getDeclaredMethod("writePathCache");m.setAccessible(true);
        for(Siding s:data.sidings)m.invoke(s);for(Siding s:data.sidings)s.init();
        if(data.stations.size()!=20||data.platforms.size()!=38||data.routes.size()!=6||data.depots.size()!=6||data.sidings.size()!=6||data.rails.size()!=manifest.getAsJsonObject("target_counts").get("rails").getAsInt())
            throw new AssertionError("Native binary patch changed record identity/count");
        return data;
    }
    private static void samePoint(org.mtr.core.tool.Vector actual,JsonArray expected,String name)
    {
        double error=Math.max(Math.abs(actual.x-expected.get(0).getAsDouble()),Math.max(Math.abs(actual.y-expected.get(1).getAsDouble()),Math.abs(actual.z-expected.get(2).getAsDouble())));
        if(error>1e-7)throw new AssertionError(name+" point changed "+error);
    }
    private static JsonArray vector(org.mtr.core.tool.Vector p)
    {JsonArray a=new JsonArray();a.add(p.x);a.add(p.y);a.add(p.z);return a;}
    private static Map<String,Rail> namedRails(Data data,JsonObject records)
    {
        Map<String,Rail> named=new HashMap<>();
        for(Rail rail:data.rails)
        {
            JsonObject r=serialize(rail);
            for(var candidate:records.getAsJsonArray("rails"))
            {
                JsonObject c=candidate.getAsJsonObject();String member=c.get("member").getAsString();
                if(member.startsWith("UNINSTALLED_CANDIDATE") && key(r.getAsJsonObject("position1"),r.getAsJsonObject("position2")).equals(key(c.getAsJsonObject("data").getAsJsonObject("position1"),c.getAsJsonObject("data").getAsJsonObject("position2"))))
                    named.put(member.substring(member.lastIndexOf('/')+1),rail);
            }
        }
        return named;
    }
    @SuppressWarnings("unchecked")
    public static void main(String[] args)throws Exception
    {
        Path output=Path.of(args[2]);if(Files.exists(output))throw new IllegalArgumentException("Keep previous result");
        JsonObject records=JsonParser.parseString(Files.readString(Path.of(args[0]))).getAsJsonObject();
        JsonObject proposal=JsonParser.parseString(Files.readString(Path.of(args[1]))).getAsJsonObject();
        Map<String,JsonObject> plans=new HashMap<>();for(var p:proposal.getAsJsonArray("all_vehicles"))plans.put(p.getAsJsonObject().get("vehicle_id").getAsString(),p.getAsJsonObject());
        Data data=args.length>3?loadBinaryPatch(Path.of(args[3])):load(records,true);
        JsonArray fleet=new JsonArray();int checked=0,affected=0;
        for(Siding siding:data.sidings)for(Vehicle vehicle:(Iterable<Vehicle>)field(siding,"vehicles"))
        {
            JsonObject plan=plans.remove(Long.toString(vehicle.getId()));if(plan==null)throw new AssertionError("Unexpected/replaced vehicle");
            JsonObject actual=serialize(vehicle);
            for(var e:plan.getAsJsonObject("preserved_state").entrySet())if(!actual.get(e.getKey()).equals(e.getValue()))throw new AssertionError("Lost original "+e.getKey());
            for(String k:List.of("railProgress","nextStoppingIndexAto","nextStoppingIndexManual"))if(!actual.get(k).equals(plan.get(k)))throw new AssertionError("Wrong mapped "+k);
            JsonArray expectedPath=plan.getAsJsonArray("new_expected_path");var path=vehicle.vehicleExtraData.immutablePath;
            if(expectedPath.size()!=path.size())throw new AssertionError("Native repeat/overlap assembly differs "+vehicle.getId()+": expected "+expectedPath.size()+" got "+path.size());
            for(int i=0;i<path.size();i++)
            {
                JsonObject e=expectedPath.get(i).getAsJsonObject(), a=serialize(path.get(i));
                for(String k:List.of("startPosition","endPosition","savedRailBaseId","dwellTime","stopIndex"))if(!a.get(k).equals(e.get(k)))throw new AssertionError("Native path differs "+vehicle.getId()+"/"+i+"/"+k);
                for(String k:List.of("startDistance","endDistance"))if(Math.abs(a.get(k).getAsDouble()-e.get(k).getAsDouble())>1e-7)
                    throw new AssertionError("Native distance differs "+vehicle.getId()+"/"+i+"/"+k+": native="+a.get(k)+" expected="+e.get(k)+" length="+path.get(i).getRailLength()+" expected_length="+e.get("native_rail_length")+" segment="+e.get("candidate_segment_id")+" cached="+data.rails.contains(path.get(i).getRail()));
                if(path.get(i).getEndDistance()<1e300 && path.get(i).getSpeedLimitKilometersPerHour()<=0)throw new AssertionError("Blocked travel direction "+vehicle.getId()+"/"+i);
            }
            int next=(int)actual.get("nextStoppingIndexAto").getAsLong();JsonObject stop=serialize(path.get(next));
            for(var e:plan.getAsJsonObject("logical_next_stop").entrySet())if(!stop.get(e.getKey()).equals(e.getValue()))throw new AssertionError("Lost logical next stop "+e.getKey());
            if(plan.has("logical_next_stop_manual")){JsonObject manualStop=serialize(path.get(actual.get("nextStoppingIndexManual").getAsInt()));for(var e:plan.getAsJsonObject("logical_next_stop_manual").entrySet())if(!manualStop.get(e.getKey()).equals(e.getValue()))throw new AssertionError("Lost manual logical next stop "+e.getKey());}
            double progress=actual.get("railProgress").getAsDouble();int current=Utilities.getIndexFromConditionalList(path,progress);
            PathData p=path.get(current);var position=p.getPosition(progress-p.getStartDistance());var head=vehicle.getHeadPosition();
            boolean movingGeometry=!plan.get("changed_original_rail_index").isJsonNull();
            if(!movingGeometry)
            {
                JsonArray old=plan.getAsJsonArray("old_native_progress_position");
                samePoint(position,old,"unchanged rail progress");
                samePoint(head,plan.getAsJsonArray("old_native_head_position"),"unchanged rail head");
            }
            if(!plan.get("affected").getAsBoolean())samePoint(head,plan.getAsJsonArray("old_native_head_position"),"unaffected fleet head");
            JsonObject row=new JsonObject();row.addProperty("vehicle_id",Long.toString(vehicle.getId()));row.addProperty("affected",plan.get("affected").getAsBoolean());
            row.addProperty("old_progress",plan.get("old_progress").getAsDouble());row.addProperty("new_progress",progress);
            row.addProperty("new_native_path_index",current);row.add("new_progress_position",vector(position));row.add("new_head_position",vector(head));
            row.add("old_progress_position",plan.get("old_native_progress_position"));row.add("old_head_position",plan.get("old_native_head_position"));
            row.addProperty("old_next_ato",plan.get("old_next_ato").getAsLong());row.addProperty("new_next_ato",next);
            row.add("logical_next_stop",plan.get("logical_next_stop"));fleet.add(row);checked++;if(plan.get("affected").getAsBoolean())affected++;
        }
        if(!plans.isEmpty()||checked!=40||affected!=6)throw new AssertionError("Fleet count/identity change: checked="+checked+", affected="+affected+", missing="+plans.keySet());
        JsonArray tests=new JsonArray();Method blocked=Rail.class.getDeclaredMethod("isBlocked",long.class,Rail.BlockReservation.class);blocked.setAccessible(true);
        for(var b:proposal.getAsJsonArray("blocks"))
        {
            Data fixture=load(records,false);Map<String,Rail> named=namedRails(fixture,records);
            JsonObject block=b.getAsJsonObject();Rail entry=named.get(block.getAsJsonArray("shared").get(0).getAsString());
            Rail opposite=named.get(block.getAsJsonArray("adjacent").get(0).getAsString());
            if((boolean)blocked.invoke(entry,111L,Rail.BlockReservation.PRE_RESERVE))throw new AssertionError("Empty block blocked first train");
            if(!(boolean)blocked.invoke(opposite,222L,Rail.BlockReservation.PRE_RESERVE))throw new AssertionError("Opposing train admitted");
            if((boolean)blocked.invoke(opposite,111L,Rail.BlockReservation.PRE_RESERVE))throw new AssertionError("Own block rejected");
            if(!(boolean)blocked.invoke(opposite,222L,Rail.BlockReservation.DO_NOT_RESERVE))throw new AssertionError("Non-reserving query failed");
            for(var member:block.getAsJsonArray("members"))
                if(!(boolean)blocked.invoke(named.get(member.getAsString()),222L,Rail.BlockReservation.DO_NOT_RESERVE))throw new AssertionError("Block member not reserved "+member);
            for(var other:proposal.getAsJsonArray("blocks"))
            {
                JsonObject o=other.getAsJsonObject();if(o.get("color").equals(block.get("color")))continue;
                Rail independent=named.get(o.getAsJsonArray("shared").get(0).getAsString());
                if((boolean)blocked.invoke(independent,333L,Rail.BlockReservation.DO_NOT_RESERVE))throw new AssertionError("Independent block coupled");
            }
            Map<String,Rail> reversed=namedRails(load(records,false),records);
            Rail reverseEntry=reversed.get(block.getAsJsonArray("adjacent").get(0).getAsString());
            if((boolean)blocked.invoke(reverseEntry,222L,Rail.BlockReservation.CURRENTLY_RESERVE))throw new AssertionError("Reverse first train rejected");
            for(var member:block.getAsJsonArray("members"))
                if(!(boolean)blocked.invoke(reversed.get(member.getAsString()),111L,Rail.BlockReservation.DO_NOT_RESERVE))throw new AssertionError("Reverse block member not reserved "+member);
            JsonObject noColors=records.deepCopy();
            for(var value:noColors.getAsJsonArray("rails"))value.getAsJsonObject().getAsJsonObject("data").add("signalColors",new JsonArray());
            Map<String,Rail> control=namedRails(load(noColors,false),noColors);
            if((boolean)blocked.invoke(control.get(block.getAsJsonArray("shared").get(0).getAsString()),111L,Rail.BlockReservation.PRE_RESERVE)
                    ||(boolean)blocked.invoke(control.get(block.getAsJsonArray("adjacent").get(0).getAsString()),222L,Rail.BlockReservation.PRE_RESERVE))throw new AssertionError("No-color negative control");
            JsonObject test=new JsonObject();test.add("color",block.get("hex_color"));test.addProperty("first_train_allowed",true);
            test.addProperty("opposing_train_blocked_at_other_entrance",true);test.addProperty("same_train_reentry_allowed",true);
            test.addProperty("other_two_groups_unblocked",true);tests.add(test);
            test.addProperty("all_members_reserved",block.getAsJsonArray("members").size());
            test.addProperty("opposite_entry_CURRENTLY_RESERVE_excludes_all_members",true);
            test.addProperty("no_signal_negative_control_admits_both",true);
        }
        JsonObject report=new JsonObject();report.addProperty("schema","projectseele.r52.native-mtr-candidate-verification.v1");
        report.addProperty("native_database_written",false);report.addProperty("simulator_started",false);
        report.addProperty("fleet_input_loader",args.length>3?"actual patched member bytes / native MessagePackReader":"uninstalled JSON proposal / native JsonReader");
        report.addProperty("vehicles_exact_original_identity_and_saved_state",checked);report.addProperty("vehicles_remapped",affected);
        report.addProperty("unaffected_vehicles_native_head_positions_exact",checked-affected);report.addProperty("native_reassembly_matches_expected_paths",true);
        report.add("fleet",fleet);report.add("native_signal_block_tests",tests);
        JsonArray members=new JsonArray();
        for(var e:namedRails(data,records).entrySet())
        {
            String id=e.getValue().getHexId();JsonObject member=new JsonObject();
            member.addProperty("segment_id",e.getKey());member.addProperty("native_hex_id",id);
            member.addProperty("native_relative_path","rails/"+id.substring(id.length()-2)+"/"+id);
            members.add(member);
        }
        report.add("candidate_native_rail_member_names",members);
        Files.writeString(output,new GsonBuilder().setPrettyPrinting().create().toJson(report)+"\n");
        System.out.println("PASS native candidate reassembly: fleet="+checked+", remapped="+affected+", native exclusive blocks="+tests.size());
    }
}
