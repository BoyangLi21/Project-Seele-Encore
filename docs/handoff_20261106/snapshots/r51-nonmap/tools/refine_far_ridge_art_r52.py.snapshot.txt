"""Root refinement of the generated ridge, preserving every operating interface.

The original native terrain is the lower bound. Authored hardware, native
access paths and the cannon/landing surfaces keep their actual levels. Only
the generated hill is reshaped, with complete originally planted trees.
"""
import os
os.environ.setdefault('OPENBLAS_NUM_THREADS','1')
import argparse,json,msvcrt
import numpy as np
from scipy.ndimage import distance_transform_edt
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import regional_voxels as vox
from r52_spatial_plan import BASE
from survey_r52_original_complex import measure
from query_blocks import iter_block_entities
from prepare_continuous_upright_civil_r52 import cuboids
from extend_world_depth_r52 import acquire_world_lock


def smooth(t):t=np.clip(t,0,1);return t*t*(3-2*t)


def main():
    ap=argparse.ArgumentParser();ap.add_argument('--apply',action='store_true');args=ap.parse_args()
    out=BASE/'final_map_audit/far_ridge_refinement';out.mkdir(exist_ok=True);stage='irregular_ridge_preserved_operating_ground'
    assert not list((out/stage).glob('applied_*')), 'Do not replay'
    design=json.loads((BASE/'far_yashima/RIDGE_DESIGN.json').read_text());source=np.load(BASE/'far_yashima/RIDGE_HEIGHTS.npz')
    lo=(-1672,32,-1852);hi=(-955,223,-1129)
    before,names=measure(lo,hi);after=before.copy()
    for state in ('minecraft:air','minecraft:stone','minecraft:dirt','minecraft:grass_block[snowy=false]',
                  'minecraft:oak_log[axis=y]','minecraft:oak_leaves[distance=1,persistent=true,waterlogged=false]'):
        if state not in names:names.append(state)
    ids={s:i for i,s in enumerate(names)}
    sx,sz=lo[0]-int(source['origin'][0]),lo[2]-int(source['origin'][1]);nz,nx=before.shape[1:]
    base=source['before'][sz:sz+nz,sx:sx+nx];owned=source['active'][sz:sz+nz,sx:sx+nx]
    z,x=np.mgrid[lo[2]:hi[2]+1,lo[0]:hi[0]+1]
    soil_names={'minecraft:stone','minecraft:dirt','minecraft:grass_block','minecraft:gravel','minecraft:sand','minecraft:clay'}
    soil=np.array([s.split('[')[0] in soil_names for s in names]);ground=np.full((nz,nx),-999,np.int16)
    authored=np.array([not vox.natural(s) for s in names]);hardware=np.zeros((nz,nx),bool)
    for i,y in enumerate(range(lo[1],hi[1]+1)):
        ground=np.where(soil[before[i]],y,ground);hardware|=authored[before[i]]
    assert np.all(ground[owned]>=32)
    # A curved watershed with several unequal shoulders replaces the three
    # independent radial bells. It stays inside the prior generated footprint.
    points=[(-1560,-1680,132,105),(-1440,-1580,195,145),(-1360,-1460,190,135),
            (-1250,-1430,184,140),(-1110,-1490,156,95),(-1030,-1420,119,70)]
    terrain=base.astype(float).copy()
    warp=7*np.sin(x/91+z/137)+4*np.sin(z/57-x/119)
    for a,b in zip(points,points[1:]):
        p=np.array(a[:2],float);d=np.array(b[:2],float)-p
        t=np.clip(((x-p[0])*d[0]+(z-p[1])*d[1])/(d@d),0,1)
        distance=np.hypot(x-(p[0]+t*d[0]),z-(p[1]+t*d[1]))
        width=a[3]*(1-t)+b[3]*t;peak=a[2]*(1-t)+b[2]*t
        profile=np.maximum(0,1-(distance/(width*1.65))**2)**2
        candidate=base+(peak-base+warp)*profile
        terrain=np.maximum(terrain,candidate)
    centre=np.array(design['centre_xz']);f=np.array(design['forward_xz']);r=np.array(design['right_xz'])
    u=(x-centre[0])*r[0]+(z-centre[1])*r[1];v=(x-centre[0])*f[0]+(z-centre[1])*f[1]
    keep=hardware.copy()
    # This strip includes the actual prone EVA, full gun service floor,
    # shield stance and their safety margin. Arrival pads stay separate.
    keep|=(abs(u)<=44)&(v>=-86)&(v<=90)
    site=json.loads((BASE/'far_yashima_access/ROOT_SITE_CANDIDATE.json').read_text())
    paths=list(site['eva_access_routes'].values())+[site['cover_after_supply_route']]
    for path in paths:
        for aa,bb in zip(path,path[1:]):
            a=np.array(aa)[[0,2]];d=np.array(bb)[[0,2]]-a;length=d@d
            if length<1e-8:continue
            t=np.clip(((x-a[0])*d[0]+(z-a[1])*d[1])/length,0,1)
            keep|=(x-a[0]-t*d[0])**2+(z-a[1]-t*d[1])**2<=18**2
    for p in design['points'].values():keep|=(x-p[0])**2+(z-p[2])**2<=16**2
    distance=distance_transform_edt(~keep);blend=1-smooth(distance/68)
    terrain=terrain*(1-blend)+ground*blend
    # Retain the original survey boundary, with a smooth transition to it.
    inward=distance_transform_edt(owned);boundary=smooth(inward/22)
    height=np.rint(np.maximum(base,ground*(1-boundary)+terrain*boundary)).astype(np.int16)
    height[keep]=ground[keep];height[~owned]=ground[~owned];height=np.minimum(height,208)
    # Raising the generated low side closes new one-column cliffs exposed
    # by lowering the former radial mound. Existing native sharp edges stay
    # measured exceptions; no new edge may be worse than the old edge or 2m.
    adjustable=owned & ~keep
    ceiling=np.full(height.shape,30000,np.int32);ceiling[~adjustable]=ground[~adjustable]
    for iteration in range(192):
        previous=ceiling.copy()
        for axis in (0,1):
            aa=(slice(None,-1),slice(None)) if axis==0 else (slice(None),slice(None,-1))
            bb=(slice(1,None),slice(None)) if axis==0 else (slice(None),slice(1,None))
            limit=np.maximum(2,np.abs(ground[aa].astype(int)-ground[bb].astype(int)))
            left=ceiling[aa].copy();right=ceiling[bb].copy();updated=ceiling.copy()
            updated[aa]=np.minimum(updated[aa],right+limit)
            updated[bb]=np.minimum(updated[bb],left+limit);ceiling=updated
        if np.array_equal(previous,ceiling):break
    assert np.all(ceiling>=base),'Protected ground conflicts with native terrain lower bound'
    height=np.minimum(height,ceiling).astype(np.int16)
    for iteration in range(128):
        previous=height.copy()
        for axis in (0,1):
            aa=(slice(None,-1),slice(None)) if axis==0 else (slice(None),slice(None,-1))
            bb=(slice(1,None),slice(None)) if axis==0 else (slice(None),slice(1,None))
            limit=np.maximum(2,np.abs(ground[aa].astype(int)-ground[bb].astype(int)))
            left=height[aa].copy();right=height[bb].copy()
            updated=height.copy()
            updated[aa]=np.where(adjustable[aa],np.maximum(updated[aa],right-limit),updated[aa])
            updated[bb]=np.where(adjustable[bb],np.maximum(updated[bb],left-limit),updated[bb])
            height=updated
        if np.array_equal(height,previous):break
    for axis in (0,1):
        limit=np.maximum(2,np.abs(np.diff(ground.astype(int),axis=axis)))
        bad=np.abs(np.diff(height.astype(int),axis=axis))>limit
        if bad.any():
            pp=np.argwhere(bad)[:15]
            print('CLIFF',axis,[(q.tolist(),int(height[tuple(q)]),int(ground[tuple(q)]),bool(keep[tuple(q)]),bool(owned[tuple(q)])) for q in pp],flush=True)
        assert not bad.any(), 'New terrain cliff remains'
    active=owned&(height!=ground)
    assert not np.any(active&hardware), 'Existing operating equipment changed'
    trees=json.loads((BASE/'far_yashima/far_ridge_native_blend/NEW_TREES.json').read_text())
    translated=[];tree_cells=[]
    for tx,ty,tz,trunk in trees:
        dz=int(height[tz-lo[2],tx-lo[0]]-ground[tz-lo[2],tx-lo[0]])
        if not dz:continue
        cy=ty-1+trunk;cells={}
        for dy,rad in ((-3,3),(-2,3),(-1,2),(0,2),(1,1)):
            for oz in range(-rad,rad+1):
                for ox in range(-rad,rad+1):
                    if ox*ox+oz*oz<=rad*rad+1:cells[tx+ox,cy+dy,tz+oz]=ids['minecraft:oak_leaves[distance=1,persistent=true,waterlogged=false]']
        for y in range(ty,cy+1):cells[tx,y,tz]=ids['minecraft:oak_log[axis=y]']
        for (px,py,pz),value in cells.items():
            assert lo[0]<=px<=hi[0] and lo[2]<=pz<=hi[2] and lo[1]<=py+dz<=hi[1]
            assert before[py-lo[1],pz-lo[2],px-lo[0]]==value,('Changed planted tree',px,py,pz)
            after[py-lo[1],pz-lo[2],px-lo[0]]=ids['minecraft:air']
        tree_cells.append((cells,dz));translated.append(dict(root=[tx,ty,tz],delta_y=dz,cells=len(cells)))
    natural_ids=[i for i,s in enumerate(names) if vox.natural(s)]
    for i,y in enumerate(range(lo[1],hi[1]+1)):
        edit=active&(y>=np.minimum(ground,height)-3)&(y<=np.maximum(ground,height))
        assert np.all(np.isin(after[i][edit],natural_ids)),('Authored content in terrain volume',y)
        target=np.where(y>height,ids['minecraft:air'],np.where(y==height,ids['minecraft:grass_block[snowy=false]'],
                   np.where(y>=height-3,ids['minecraft:dirt'],ids['minecraft:stone'])))
        after[i][edit]=target[edit]
    for cells,dy in tree_cells:
        for (px,py,pz),value in cells.items():
            q=(py+dy-lo[1],pz-lo[2],px-lo[0]);old=names[int(after[q])]
            assert old in ('minecraft:air','minecraft:grass','minecraft:fern') or 'leaves' in old or 'log' in old,('Moved canopy hits terrain/equipment',px,py+dy,pz,old)
            after[q]=value
    devices=dict(iter_block_entities(BASE/'WORLD_R52','projectseele:geofront',lo,hi))
    for px,py,pz in devices:assert before[py-lo[1],pz-lo[2],px-lo[0]]==after[py-lo[1],pz-lo[2],px-lo[0]]
    fig,axs=plt.subplots(1,2,figsize=(14,8),layout='constrained')
    for ax,h,title in zip(axs,[ground,height],['Current saved radial hills and oval shelf','Proposed ridge: native interfaces protected']):
        ax.imshow(h,origin='lower',extent=[lo[0],hi[0],lo[2],hi[2]],cmap='terrain',vmin=40,vmax=208)
        ax.contour(x,z,h,levels=[100,140,174,190],colors='black',linewidths=.4)
        ax.contour(x,z,keep,levels=[.5],colors='red',linewidths=.4);ax.set(title=title,xlabel='X',ylabel='Z')
    fig.savefig(out/'RIDGE_BEFORE_AFTER.png',dpi=135);plt.close(fig)
    vox.WORLD=BASE/'WORLD_R52';vox.OUT=out;vox.DIM='projectseele:geofront';p=vox.Painter(vertical_bounds=(-960,320))
    n=cuboids(p,lo,before,after,names,'R52_TV_landscape_ridge_refinement')
    p.meta.update(root_author=True,full_existing_terrain_lower_bound=True,operating_paths_unchanged=True,all_native_devices_preserved=True)
    folder=p.save_plan(stage);report=dict(world_written=False,changed_cells=n,changed_columns=int(active.sum()),
        moved_complete_trees=translated,protected_devices=len(devices),protected_ground_columns=int(keep.sum()),native_art_review_pending=True)
    np.savez_compressed(out/'REVIEW_HEIGHTS.npz',origin=[lo[0],lo[2]],before=ground,after=height,active=active,keep=keep)
    if args.apply:
        lock=acquire_world_lock()
        try:report.update(world_written=True,receipt=p.apply(stage,session_lock=lock))
        finally:msvcrt.locking(lock.fileno(),msvcrt.LK_UNLCK,1);lock.close()
    (folder/'PLAN.json').write_text(json.dumps(report,indent=2),'utf8')
    print(json.dumps({k:v for k,v in report.items() if k!='moved_complete_trees'}),flush=True)


if __name__=='__main__':main()
