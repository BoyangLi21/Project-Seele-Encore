"""Render Root's exact current/proposed civil and select finite native seats.

Selections are proposals until the map application receipt and exact native
readback exist. No world writes or production-enable flag are emitted here.
"""
import json, hashlib
from collections import defaultdict, Counter
from pathlib import Path
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from query_blocks import iter_selected_sections
from r52_spatial_plan import BASE

ROOT=Path(__file__).resolve().parents[1]
OUT=BASE/'continuous_upright/civil_replacement'


def main():
    src=np.load(OUT/'CIVIL_CANDIDATE.npz');keys=src['section_keys'];before=src['before'];after=src['after'];pal=list(map(str,src['palette']))
    index={tuple(map(int,k)):i for i,k in enumerate(keys)}
    def sample(points,which=after):
        points=np.floor(points).astype(int);result=[]
        for x,y,z in points:
            k=index.get((x//16,z//16,y//16));j=(y%16)*256+(z%16)*16+x%16
            result.append(-1 if k is None else int(which[k,j]))
        return np.array(result)
    colours=[]
    for state in pal:
        if state in ('minecraft:air','minecraft:cave_air','minecraft:void_air'):c=[.08,.11,.13]
        elif state=='minecraft:stone' or 'deepslate_bricks' in state:c=[.32,.31,.30]
        elif 'light' in state and 'glass' not in state:c=[.8,.9,.87]
        elif 'glass' in state:c=[.3,.65,.74]
        elif 'barrier' in state:c=[.85,.24,.25]
        elif 'shaft_panel' in state:c=[.48,.57,.51]
        elif 'machine_edge' in state:c=[.23,.29,.26]
        elif 'grass_block' in state:c=[.25,.55,.25]
        elif 'dirt' in state:c=[.45,.32,.20]
        else:c=[.58,.57,.50]
        colours.append(c)
    colours=np.array(colours+[[.7,.0,.7]])
    fig,axs=plt.subplots(2,2,figsize=(18,10),layout='constrained')
    yy,zz=np.mgrid[-656:-405,-112:153];points=np.stack([np.full_like(yy,30),yy,zz],axis=-1)
    for ax,data,title in zip(axs[0],[before,after],['Bank1 previous, X=30','Bank1 continuous upright, X=30']):
        states=sample(points.reshape(-1,3),data).reshape(yy.shape)
        ax.imshow(colours[states],origin='lower',extent=[-112,152,-656,-406],aspect='equal')
        ax.set(title=title,xlabel='Z',ylabel='Y');ax.axhline(-478,color='yellow',lw=.5)
    # Longitudinal section follows the bank3 centre lane. It is a sampled
    # drawing, never substituted for the complete convex-envelope proof.
    ss=np.arange(-40,2361,2);y=np.arange(-660,-460);Y,S=np.meshgrid(y,ss,indexing='ij')
    vector=np.array([-1200.,1932.]);vector/=np.linalg.norm(vector)
    points=np.stack([30.5+S*vector[0],Y,-781.724+S*vector[1]],axis=-1)
    for ax,data,title in zip(axs[1],[before,after],['Bank3 previous centre section','Bank3 continuous upright centre section']):
        states=sample(points.reshape(-1,3),data).reshape(Y.shape)
        ax.imshow(colours[states],origin='lower',extent=[ss[0],ss[-1],y[0],y[-1]],aspect='auto')
        ax.set(title=title,xlabel='Distance along horizontal centreline / m',ylabel='Y')
    fig.savefig(OUT/'CIVIL_BEFORE_AFTER.png',dpi=130);plt.close(fig)
    requests_path=ROOT/'.Codex/r52_continuous_route_migration_20261012/D31_61C9_UPRIGHT_STEEL_UNANCHORED.SOCKETS.json'
    requests=json.loads(requests_path.read_text())
    offsets=np.indices((11,11,11)).reshape(3,-1).T-5
    bearing_names={'minecraft:stone','minecraft:deepslate','minecraft:deepslate[axis=y]',
        'minecraft:black_concrete','minecraft:iron_block','minecraft:deepslate_bricks',
        'projectseele:nerv_shaft_panel','projectseele:nerv_machine_edge','projectseele:nerv_machine_panel',
        'projectseele:nerv_structural_panel','projectseele:nerv_floor_panel','projectseele:nerv_wall_panel'}
    bearing_ids={i for i,s in enumerate(pal) if s in bearing_names}
    seats=[];pending=[]
    for row in requests['sockets']:
        origin=np.array(row['rail_bearing']);blocks=np.floor(origin).astype(int)+offsets
        states=sample(blocks);choices=[]
        for p,state in zip(blocks,states):
            if int(state) not in bearing_ids:continue
            for axis in range(3):
                for sign in (-1,1):
                    face=np.clip(origin,p+.11,p+.89);face[axis]=p[axis]+(sign>0)
                    normal=np.zeros(3);normal[axis]=sign
                    delta=origin-face;distance=float(np.linalg.norm(delta))
                    if distance<.15 or distance>3.98 or np.dot(delta,normal)<.12:continue
                    # Require every interior sample of the short attachment
                    # to be in air, not inside an unrelated wall or device.
                    choices.append((distance,tuple(map(int,p)),pal[int(state)],face.tolist(),normal.tolist()))
        choice=None
        for candidate in sorted(choices,key=lambda c:c[0]):
            face=np.array(candidate[3])
            interior=face[None,:]+(origin-face)[None,:]*np.linspace(.04,.97,24)[:,None]
            values=sample(interior)
            if all(v>=0 and pal[int(v)] in ('minecraft:air','minecraft:cave_air','minecraft:void_air') for v in values):
                choice=candidate;break
        if choice is None:
            pending.append(row);continue
        seats.append(dict(socket_id=row['socket_id'],at=choice[1],state=choice[2],bearing=choice[3],normal=choice[4],
                          connector_distance_m=choice[0]))
    result=dict(schema='projectseele.r52.upright-native-seat-selection.v1',status='PROPOSED_MAP_NOT_APPLIED',
        world_written=False, model_source_sha256=requests['model_source_sha256'],
        directory_geometry_sha256=requests['directory_geometry_sha256'],
        civil_candidate_sha256=hashlib.sha256((OUT/'CIVIL_CANDIDATE.npz').read_bytes()).hexdigest(),seats=seats,pending=pending)
    (OUT/'PROPOSED_NATIVE_SEATS.json').write_text(json.dumps(result,indent=2),'utf8')
    print(json.dumps(dict(seats=len(seats),pending=len(pending),max_distance=max(s['connector_distance_m'] for s in seats),
        pending_by_leg=dict(Counter(p['leg'] for p in pending)))),flush=True)


if __name__=='__main__':main()
