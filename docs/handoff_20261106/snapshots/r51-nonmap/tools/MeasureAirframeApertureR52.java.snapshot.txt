package com.projectseele.world;
import com.google.gson.*;
import com.projectseele.entity.*;
import java.nio.file.*;
import java.lang.reflect.*;
import java.util.*;
import org.joml.Quaternionf;
import org.joml.Vector3f;
import net.minecraft.world.phys.*;
import com.projectseele.world.NervAirTransportManifestR52.*;

/** Source vertex measurement only. This neither chooses nor installs a shaft. */
public final class MeasureAirframeApertureR52
{
    private static JsonObject json(Path p)throws Exception{return JsonParser.parseString(Files.readString(p)).getAsJsonObject();}
    private static Map<String,String> hashes(JsonObject o){var m=new HashMap<String,String>();o.getAsJsonObject("resources").entrySet().forEach(e->m.put(e.getKey(),e.getValue().getAsString()));return m;}
    private static Vec3 read(JsonArray a){return new Vec3(a.get(0).getAsDouble(),a.get(1).getAsDouble(),a.get(2).getAsDouble());}
    private static JsonArray vector(Vec3 p){var a=new JsonArray();a.add(p.x);a.add(p.y);a.add(p.z);return a;}
    private static JsonArray box(AABB p){var a=vector(new Vec3(p.minX,p.minY,p.minZ));a.add(p.maxX);a.add(p.maxY);a.add(p.maxZ);return a;}
    private static EvaBodyPose.Sample pose(JsonObject root,int variant)
    {
        var rig=new LinkedHashMap<String,EvaBodyPose.Bone>();for(var raw:root.getAsJsonObject("rigs").getAsJsonArray(Integer.toString(variant))){var row=raw.getAsJsonObject();var p=row.getAsJsonArray("pivot");var pivot=new Vector3f(p.get(0).getAsFloat(),p.get(1).getAsFloat(),p.get(2).getAsFloat()).mul(-1,1,1).div(16);var r=row.has("rotation")?row.getAsJsonArray("rotation"):row.getAsJsonArray("bindRotationDegrees");var q=new Quaternionf();if(r!=null)q.rotationZYX(r.get(2).getAsFloat()*(float)Math.PI/180,-r.get(1).getAsFloat()*(float)Math.PI/180,-r.get(0).getAsFloat()*(float)Math.PI/180);String name=row.get("name").getAsString(),parent=row.has("parent")&&!row.get("parent").isJsonNull()?row.get("parent").getAsString():null;rig.put(name,new EvaBodyPose.Bone(name,parent,pivot,q));}var pose=new EvaBodyPose.Sample(Map.copyOf(rig));rig.values().forEach(b->pose.rotations.put(b.name(),new Quaternionf(b.bindRotation())));pose.dirty();return pose;
    }
    private static AABB bounds(double[] xyz,RigidTransform frame){AABB b=null;for(int i=0;i<xyz.length;i+=3){var p=frame.transformPoint(new Vec3(xyz[i],xyz[i+1],xyz[i+2]));var q=new AABB(p,p);b=b==null?q:b.minmax(q);}return b;}
    public static void main(String[] args)throws Exception
    {
        var config=json(Path.of(args[0]));var bodyData=json(Path.of(args[1]));var provider=EvaOriginalMeshGeometryR52.load(EvaOriginalMeshGeometryR52::classpathBytes,hashes(config));var full=FullAssemblyGeometryR52.load(EvaOriginalMeshGeometryR52::classpathBytes,hashes(json(Path.of(args[2]))));var aeroHashes=new HashMap<String,String>();for(String path:List.of("mesh/tv_facilities_r16.json","mesh/air_cradle_r31.json")){var bytes=EvaOriginalMeshGeometryR52.classpathBytes(path);aeroHashes.put(path,HexFormat.of().formatHex(java.security.MessageDigest.getInstance("SHA-256").digest(bytes)));}var aero=NervAirframeGeometryR52.loadClasspath(aeroHashes);
        var controls=EvaAnatomicalHandsR45.class.getDeclaredMethod("controls",EvaAnatomicalHandsR45.Rig.class,String.class,String.class);controls.setAccessible(true);var rotation=EvaAnatomicalHandsR45.class.getDeclaredMethod("rotation",EvaAnatomicalHandsR45.Joint.class,float.class,float.class,float.class);rotation.setAccessible(true);var output=new JsonObject();output.addProperty("schema","projectseele.r52.actual-aircraft-aperture-measurement.v1");output.addProperty("scope","Actual source vertices; upright factory and explicit rigid-prone reference. No frozen field HULL, map installation or native acceptance claim.");output.addProperty("hoist",112);output.addProperty("plane_model_yaw_degrees",0);output.addProperty("aircraft_source",aero.fingerprint());var sourceRows=new JsonObject();aeroHashes.forEach(sourceRows::addProperty);output.add("aircraft_mesh_sources",sourceRows);var dorsal=json(Path.of(args[4])).getAsJsonObject("profiles");var rows=new JsonArray();
        for(int variant=0;variant<3;variant++)
        {
            var body=pose(bodyData,variant);var rig=EvaAnatomicalHandsR45.rig(variant);var raw=(Map<String,float[]>)controls.invoke(null,rig,"relaxed","relaxed");var hands=new HashMap<String,Quaternionf>();for(var j:rig.joints()){var v=raw.get(j.name());hands.put(j.name(),(Quaternionf)rotation.invoke(null,j,v[0],v[1],v[2]));}var geometry=provider.evaluate(variant,body,new EvaAnatomicalHandsR45.Pose("relaxed","relaxed",hands));
            var targets=new ArrayList<Vec3>();for(String bone:List.of("arm_l","arm_r","leg_l","leg_r","shin_l","shin_r")){var p=body.matrix(bone).transformPosition(new Vector3f(body.rig.get(bone).pivot()).add(0,0,2.5F/EvaScale.RENDER_SCALE)).mul(EvaScale.RENDER_SCALE);targets.add(new Vec3(p.x,p.y,p.z));}var contacts=NervAirCradleContactsR52.evaluate(geometry,targets);var centres=contacts.stream().map(NervAirCradleContactsR52.Contact::centre).toList();var axes=contacts.stream().map(NervAirCradleContactsR52.Contact::axis).toList();
            var d=dorsal.getAsJsonObject("eva_unit0"+variant);var profile=new EvaDorsalProfile.Profile(read(d.getAsJsonArray("centre")),read(d.getAsJsonArray("outward")),read(d.getAsJsonArray("hinge")),read(d.getAsJsonArray("hinge_axis")),d.get("open_angle_degrees").getAsFloat());var socket=EvaMechanicalFrameMathR52.emptySocket(body,EvaRifleKinematics.frameWorldR52(Vec3.ZERO,180),profile);var plugParts=full.closedPlugParts(variant);
            for(String posture:List.of("upright","rigid_prone_x90_reference"))for(boolean carrier:List.of(false,true))
            {
                var cargo=RigidTransform.of(new Vec3(0,-112,0),new Quaternionf().rotationY((float)Math.PI).rotateX(posture.equals("upright")?0:(float)Math.PI/2));var frame=new Frame(RigidTransform.identity(),cargo,new Rig(112,1,0));AABB all=null;for(var part:aero.parts(frame,centres,axes,0)){var b=bounds(part.xyz(),RigidTransform.identity());all=all==null?b:all.minmax(b);}for(var p:geometry.parts())all=all.minmax(bounds(p.triangles(),cargo));for(var p:plugParts.values())all=all.minmax(bounds(p,cargo.compose(socket)));if(carrier)for(var p:full.carrierParts(1,false,0).values())all=all.minmax(bounds(p,cargo));
                var row=new JsonObject();row.addProperty("variant",variant);row.addProperty("body_source",geometry.sourceFingerprint());row.addProperty("posture",posture);row.addProperty("actual_rise1_carrier_included",carrier);row.add("exact_source_vertex_union",box(all));row.addProperty("minimum_x_span",all.getXsize());row.addProperty("minimum_z_span",all.getZsize());row.addProperty("lowest_below_plane",-all.minY);row.addProperty("highest_above_plane",all.maxY);
                var actualParts=aero.parts(frame,centres,axes,0);var contactRows=new JsonArray();for(int index=0;index<6;index++){var contact=contacts.get(index);var skin=cargo.transformPoint(contact.skin());String jawName="jaw/"+(index/2)+"/"+(index%2);var jaw=actualParts.stream().filter(p->p.name().equals(jawName)).findFirst().orElseThrow();var xyz=jaw.xyz();double distance=Double.POSITIVE_INFINITY;for(int n=0;n<xyz.length;n+=9){var a=new Vec3(xyz[n],xyz[n+1],xyz[n+2]);var b=new Vec3(xyz[n+3],xyz[n+4],xyz[n+5]);var c=new Vec3(xyz[n+6],xyz[n+7],xyz[n+8]);if(b.subtract(a).cross(c.subtract(a)).lengthSqr()>1e-20)distance=Math.min(distance,NervAirCradleContactsR52.closest(skin,a,b,c).distanceToSqr(skin));}if(distance>1e-6)throw new IllegalStateException("Actual skin/jaw contact lost: "+variant+"/"+posture+"/"+contact.bone()+" "+distance);var detail=new JsonObject();detail.addProperty("bone",contact.bone());detail.addProperty("actual_skin_to_actual_jaw_distance_metres",Math.sqrt(distance));contactRows.add(detail);}row.add("actual_six_skin_jaw_contacts",contactRows);
                var collisions=new JsonArray();for(int index=0;index<5;index++){var aircraftMesh=new NervRigidMeshSweepR52(actualParts.get(index).xyz());for(var bodyPart:geometry.parts()){var bodyMesh=new NervRigidMeshSweepR52(bodyPart.triangles());if(aircraftMesh.intersects(RigidTransform.identity(),RigidTransform.identity(),bodyMesh,cargo,cargo,.005,Vec3.ZERO)){collisions.add(actualParts.get(index).name()+"/body/"+bodyPart.bone());break;}}for(var plugPart:plugParts.entrySet()){var plugMesh=new NervRigidMeshSweepR52(plugPart.getValue());var transform=cargo.compose(socket);if(aircraftMesh.intersects(RigidTransform.identity(),RigidTransform.identity(),plugMesh,transform,transform,.005,Vec3.ZERO)){collisions.add(actualParts.get(index).name()+"/closed_capsule/"+plugPart.getKey());break;}}}row.add("actual_aircraft_body_rotor_vs_cargo_skin_crossings",collisions);rows.add(row);
            }
        }
        output.add("cases",rows);Files.writeString(Path.of(args[3]),new GsonBuilder().setPrettyPrinting().create().toJson(output));System.out.println(output);
    }
}
