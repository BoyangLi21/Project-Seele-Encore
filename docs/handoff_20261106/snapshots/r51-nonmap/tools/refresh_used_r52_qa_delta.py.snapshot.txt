"""Root-only recorded final-baseline refresh of the already used R52 QA.

Prepare a bounded member delta after actual Source install/save/cold restart.
Apply and rollback require the exact reviewed plan SHA. Never copy a client,
another save, level.dat or player progress; never write anything back to Source.
"""

from pathlib import Path, PurePosixPath
from build_immutable_r52_candidate import frozen_build_class_pins

ROOT=Path(__file__).resolve().parents[1]
BASE=Path('D:/eva/artifacts/r52_fresh_from_r51_20261009').resolve()
SOURCE=BASE/'WORLD_R52'
GAME=Path('D:/eva/artifacts/r51_nonmap_release/native_r51/game').resolve()
QA=GAME/'saves/SEELE_R52_REVIEW'
CLONE=BASE/'native_r52_validation/CLONE_RECEIPT.json'
DATA_NAMES={'projectseele_dead_sea_chamber_r45.dat','projectseele_facility_v2_lifts_e2.dat','projectseele_facility_v2_e2.dat','projectseele_facility_v2_command_interior_e2.dat','projectseele_equipment_vaults_r47.dat','projectseele_nerv_routes_r52.dat'}
ROOT_JSON={'r47_seele_conference.json','r48_seele_lighting.json','quality_walk_cases.json','native_collision_shapes.json','nerv_routes_r24.json.gz','nearest_lift_paths_r44.json.gz'}
PROTECTED={'playerdata','stats','advancements','level.dat','level.dat_old','session.lock'}
def need(p,s):
    if not p:raise ValueError(s)
def sha(p):
    with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def read(p):return json.loads(Path(p).read_text('utf-8-sig'))
def write(p,d):Path(p).write_text(json.dumps(d,ensure_ascii=False,indent=2)+'\n',encoding='utf8')
def rel(name):
    p=PurePosixPath(name);need(name and not p.is_absolute() and ':' not in name and '\\' not in name and all(x not in ('','.','..') for x in name.split('/')),'Bounded relative member required');return p
def protected(name):return any(x in PROTECTED for x in rel(name).parts) or 'mtr' in name.lower()
def eligible(name,explicit):
    if protected(name):return False
    p=rel(name)
    if p.name.endswith('.mca') and p.parent.name in ('region','entities'):return True
    if p.parent.name=='data':return p.name in DATA_NAMES or re.fullmatch(r'map_\d+\.dat',p.name) is not None or name in explicit
    return name in explicit or p.parts[0]=='r52' and p.suffix=='.json' or len(p.parts)==1 and p.name in ROOT_JSON
def inventory(folder):
    rows={}
    for p in folder.rglob('*'):
        need(not p.is_symlink() and not p.is_junction(),'Linked member is not admitted: '+str(p))
        if p.is_file():rows[p.relative_to(folder).as_posix()]=sha(p)
    return rows
def require_target_hosts_idle():
    r=subprocess.run(['powershell','-NoProfile','-Command',"@(Get-CimInstance Win32_Process -Filter \"Name='java.exe'\" | Select-Object ProcessId,CommandLine) | ConvertTo-Json -Compress"],capture_output=True,text=True,check=True)
    rows=json.loads(r.stdout) if r.stdout.strip() else [];rows=rows if isinstance(rows,list) else [rows];busy=[]
    for row in rows:
        command=(row.get('CommandLine') or '').replace('\\','/').lower()
        if any(value in command for value in (GAME.as_posix().lower(),SOURCE.as_posix().lower(),QA.name.lower(),'--world world_r52')):busy.append(row['ProcessId'])
    need(not busy,'Target QA/client/Source host still running; close the relevant host normally. PIDs='+str(busy))
def locks():
    held=[]
    try:
        for world in (SOURCE,QA):
            p=world/'session.lock';need(p.is_file(),'Existing World lock required: '+str(p));f=p.open('r+b');msvcrt.locking(f.fileno(),msvcrt.LK_NBLCK,1);held.append(f)
        return held
    except:
        for f in held:msvcrt.locking(f.fileno(),msvcrt.LK_UNLCK,1);f.close()
        raise
def unlock(held):
    for f in reversed(held):f.seek(0);msvcrt.locking(f.fileno(),msvcrt.LK_UNLCK,1);f.close()
def target(row):
    base={'world':QA,'runtime':GAME/'projectseele-local-maps','jar':GAME/'mods'}[row['area']]
    p=(base/rel(row['name']).as_posix()).resolve();need(p.is_relative_to(base.resolve()) and not p.is_relative_to(SOURCE),'Exact QA/client bounded destination required')
    if row['area']=='world':need(not protected(row['name']),'Player/level/traffic progress is protected')
    if row['area']=='jar':need(p.parent==GAME/'mods' and p.name.startswith('projectseele-') and p.suffix=='.jar','Only actual mod Jar may be changed')
    return p
def final_native_binding(result_path,build_receipt,jar,jar_sha256,deployment,runtime_after,class_pins):
    native=read(result_path);inputs_path=Path(result_path).parent/'INPUTS.json';inputs=read(inputs_path)
    for name in ('installation_accepted','native_restart_passed','normal_stop_complete','actual_controller_uuid_and_sources_retained','geofront_original_identities_retained','selected_wet_floor_seal_specs_retained'):
        need(native.get(name)is True,'Final actual Source native check incomplete: '+name)
    need(native.get('pid')is None and not native.get('failure') and not native.get('live_pid_needs_root_attention'),'Actual Source native attempt failed or is still running')
    need(inputs.get('prepared_only')is False and inputs.get('no_world_copy_or_qa_import')is True and Path(native['world']).resolve()==Path(inputs['world']).resolve()==SOURCE,'Actual uncopied Source installation/save/cold restart required')
    need(Path(inputs['jar']).resolve()==Path(jar).resolve() and inputs['jar_sha256']==jar_sha256,'QA selected Jar differs from actual native install')
    need(Path(inputs['build_receipt']).resolve()==Path(build_receipt).resolve() and inputs['build_receipt_sha256']==sha(build_receipt) and inputs['compiled_delivery_class_sha256']==class_pins,'QA selected build/classes differ from actual native install')
    need(inputs['deployment_sha256']==sha(deployment),'QA final deployment differs from actual native install')
    need(inputs['runtime_sha256']==runtime_after,'QA portable runtime differs from actual native install')
    need(native['expected_controllers']==inputs['expected_controllers'] and native['expected_controller_count']==inputs['expected_controller_count']==len(inputs['expected_controllers']),'Actual cold controller manifest differs')
    return {'native_install_inputs':str(inputs_path.resolve()),'native_install_inputs_sha256':sha(inputs_path),'native_install_result':str(Path(result_path).resolve()),'native_install_result_sha256':sha(result_path),'actual_controller_count':native['expected_controller_count']}
def prepare(a):
    out=a.out.resolve();need(out.is_relative_to(BASE/'native_r52_validation') and not out.exists(),'Fresh external QA-delta evidence directory required')
    clone=read(CLONE);need(Path(clone['source_world']).resolve()==SOURCE and Path(clone['target_world']).resolve()==QA and clone['ever_launched'] is True,'Reuse actual previously launched QA, never make a new clone')
    need(sha(a.jar)==a.jar_sha256,'Selected actual Jar SHA differs');pins=frozen_build_class_pins(a.build_receipt,a.jar)
    deployment=read(a.deployment);need(deployment.get('ready') and Path(deployment['world']).resolve()==SOURCE,'Exact final Root deployment required')
    runtime=a.runtime.resolve();need(runtime.is_dir() and (runtime/'combat_bundle_r44.json').is_file(),'Complete selected portable runtime required');runtime_after=inventory(runtime)
    contract=read(runtime/'combat_bundle_r44.json');need(set(runtime_after)==set(contract['files'])|{'combat_bundle_r44.json'} and all(runtime_after.get(n)==h for n,h in contract['files'].items()),'Exact portable combat contract and members required')
    native_binding=final_native_binding(a.native_install_result,a.build_receipt,a.jar,a.jar_sha256,a.deployment,runtime_after,pins)
    explicit={r['world_relative'] for r in deployment['files']}|set(a.metadata_member)|set(a.data_member)
    for name in explicit:need(not protected(name),'Do not request protected QA member: '+name)
    require_target_hosts_idle();held=locks()
    try:
        source=inventory(SOURCE);qa=inventory(QA);rows=[]
        for name,digest in source.items():
            if eligible(name,explicit) and qa.get(name)!=digest:rows.append(dict(area='world',name=name,source=str(SOURCE/name),source_sha256=digest,before_sha256=qa.get(name),after_sha256=digest,initial_clone_sha256=clone['target_files'].get(name)))
        # A removal is only a recorded Source retirement, never a QA-only file.
        for name in clone['target_files']:
            if name not in source and name in qa and eligible(name,explicit):rows.append(dict(area='world',name=name,source=None,source_sha256=None,before_sha256=qa[name],after_sha256=None,initial_clone_sha256=clone['target_files'][name]))
        runtime_before=inventory(GAME/'projectseele-local-maps')
        for name in sorted(set(runtime_before)|set(runtime_after)):
            if runtime_before.get(name)!=runtime_after.get(name):rows.append(dict(area='runtime',name=name,source=str(runtime/name) if name in runtime_after else None,source_sha256=runtime_after.get(name),before_sha256=runtime_before.get(name),after_sha256=runtime_after.get(name)))
        for jar in (GAME/'mods').glob('projectseele-*.jar'):
            if jar.name!='projectseele-0.1.0-all.jar':rows.append(dict(area='jar',name=jar.name,source=None,source_sha256=None,before_sha256=sha(jar),after_sha256=None))
        current=GAME/'mods/projectseele-0.1.0-all.jar';before=sha(current) if current.is_file() else None
        if before!=a.jar_sha256:rows.insert(0,dict(area='jar',name=current.name,source=str(a.jar.resolve()),source_sha256=a.jar_sha256,before_sha256=before,after_sha256=a.jar_sha256))
        protected_before={n:h for n,h in qa.items() if protected(n)}
        for row in rows:target(row)
        out.mkdir(parents=True);plan=dict(schema='projectseele.r52.used-qa-final-baseline-delta.v1',prepared_at=datetime.datetime.now().isoformat(),source_world=str(SOURCE),target_world=str(QA),source_world_written=False,qa_written=False,ever_launched_preserved=True,continuity='Explicit final-baseline refresh of an already used QA; previous native sessions are historical, not uninterrupted continuation.',clone_receipt_sha256=sha(CLONE),native_install_result=str(a.native_install_result.resolve()),native_install_result_sha256=sha(a.native_install_result),deployment_sha256=sha(a.deployment),build_receipt=str(a.build_receipt.resolve()),build_receipt_sha256=sha(a.build_receipt),jar_sha256=a.jar_sha256,class_pins=pins,source_inventory=source,qa_protected_before=protected_before,runtime_before=runtime_before,runtime_after=runtime_after,changes=rows,scope='Changed selected static region/entities/data and facilities metadata only; region files are replaced as members, so QA-only block/actor edits inside those changed members are explicitly retired into this final baseline. Player/level and unselected SavedData/traffic remain untouched.',rollback_required=True,native_visual_acceptance='ROOT_NATIVE_ART_REVIEW_PENDING')
        plan.update(native_binding);write(out/'PLAN.json',plan);print(json.dumps(dict(prepared_only=True,changes=len(rows),plan=str(out/'PLAN.json'),plan_sha256=sha(out/'PLAN.json'),qa_written=False)))
    finally:unlock(held)
def apply(a,rollback=False):
    p=a.plan.resolve();need(p.parent.is_relative_to(BASE/'native_r52_validation') and sha(p)==a.plan_sha256,'Exact reviewed external plan/SHA required');d=read(p);need(Path(d['source_world']).resolve()==SOURCE and Path(d['target_world']).resolve()==QA,'Only one actual used QA and one read-only Source permitted');require_target_hosts_idle();held=locks()
    try:
        source_at_operation=inventory(SOURCE)
        if rollback:need((p.parent/'BACKUPS_READY.json').is_file() and not (p.parent/'ROLLED_BACK.json').exists(),'Complete before backups are required for full or partial rollback')
        else:
            need(source_at_operation==d['source_inventory'],'Source changed since final baseline plan; re-prepare after Root final edits')
            need(inventory(GAME/'projectseele-local-maps')==d['runtime_before'],'QA runtime changed since reviewed plan')
            need(not (p.parent/'APPLIED.json').exists(),'Delta already applied; preserve actual session evidence')
            need(sha(CLONE)==d['clone_receipt_sha256'] and sha(d['native_install_result'])==d['native_install_result_sha256'] and sha(d['build_receipt'])==d['build_receipt_sha256'],'Frozen final installation/build/clone evidence changed')
            need(sha(d['native_install_inputs'])==d['native_install_inputs_sha256'],'Final actual native input selection changed')
        protected_now={n:h for n,h in inventory(QA).items() if protected(n)}
        if not rollback:need(protected_now==d['qa_protected_before'],'Protected QA player/level/traffic files changed; do not overwrite')
        rows=list(reversed(d['changes'])) if rollback else d['changes']
        for row in rows:
            dst=target(row);current=sha(dst) if dst.is_file() else None;allowed=(row['before_sha256'],row['after_sha256']) if rollback else (row['before_sha256'],);need(current in allowed,'Actual QA member changed since plan: '+str(dst))
            src=p.parent/'before'/row['area']/row['name'] if rollback else Path(row['source']) if row['source'] else None
            digest=row['before_sha256'] if rollback else row['after_sha256'];need(digest is None or src is not None and src.is_file() and sha(src)==digest,'Exact frozen copy/rollback bytes absent')
        # Backup every changed member before the first QA mutation.
        if not rollback:
            for row in rows:
                dst=target(row)
                if dst.is_file():b=p.parent/'before'/row['area']/row['name'];b.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(dst,b);need(sha(b)==row['before_sha256'],'Complete before backup failed')
            write(p.parent/'BACKUPS_READY.json',dict(plan_sha256=a.plan_sha256,complete_before_backups=True,qa_written=False))
        log=p.parent/('ROLLBACK.jsonl' if rollback else 'APPLY.jsonl')
        for row in rows:
            dst=target(row);digest=row['before_sha256'] if rollback else row['after_sha256'];src=p.parent/'before'/row['area']/row['name'] if rollback else Path(row['source']) if row['source'] else None
            if digest is None:
                if dst.exists():dst.unlink()
            else:
                dst.parent.mkdir(parents=True,exist_ok=True);tmp=dst.with_name(dst.name+'.r52-delta-tmp');need(not tmp.exists(),'Preserve unexpected temporary member');shutil.copyfile(src,tmp);need(sha(tmp)==digest,'Actual copied bytes differ');os.replace(tmp,dst)
            with log.open('a',encoding='utf8') as f:f.write(json.dumps(dict(area=row['area'],name=row['name'],before=row['after_sha256'] if rollback else row['before_sha256'],after=digest,time=datetime.datetime.now().isoformat()))+'\n')
        need(inventory(SOURCE)==source_at_operation and {n:h for n,h in inventory(QA).items() if protected(n)}==protected_now,'Source or protected QA changed during refresh/rollback')
        need(inventory(GAME/'projectseele-local-maps')==(d['runtime_before'] if rollback else d['runtime_after']),'Actual client runtime differs from complete selected/rolled-back bytes')
        result=dict(qa_written=True,source_world_written=False,continuity=d['continuity'],changes=len(rows),plan_sha256=a.plan_sha256,rollback=rollback,player_stats_advancements_level_and_traffic_preserved=True,native_visual_acceptance='ROOT_NATIVE_ART_REVIEW_PENDING');write(p.parent/('ROLLED_BACK.json' if rollback else 'APPLIED.json'),result);print(json.dumps(result))
    finally:unlock(held)
def main():
    ap=argparse.ArgumentParser(description=__doc__);s=ap.add_subparsers(dest='mode',required=True);p=s.add_parser('prepare')
    for name in ('build-receipt','jar','runtime','deployment','native-install-result','out'):p.add_argument('--'+name,type=Path,required=True)
    p.add_argument('--jar-sha256',required=True);p.add_argument('--metadata-member',action='append',default=[]);p.add_argument('--data-member',action='append',default=[])
    for mode in ('apply','rollback'):
        p=s.add_parser(mode);p.add_argument('--plan',type=Path,required=True);p.add_argument('--plan-sha256',required=True)
    a=ap.parse_args()
    if a.mode=='prepare':prepare(a)
    else:apply(a,a.mode=='rollback')
if __name__=='__main__':main()
