"""Three bounded dry service pits beneath the unchanged wet hangar floors."""
import argparse,gzip,hashlib,json,math,uuid
from pathlib import Path
import numpy as np
import regional_voxels as vox
from r52_spatial_plan import BASE
from survey_r52_original_complex import measure
from prepare_terminal_move_r52 import runs
from query_blocks import iter_block_entities


def main():
    raise RuntimeError("Retired by user R53: restore R50 carrier; underground carrier storage pits must not be generated")
    ap=argparse.ArgumentParser();ap.add_argument('--apply',action='store_true');args=ap.parse_args()
    out=BASE/'carrier_service_pits';stage='three_dry_pits_below_original_wet_floors';plan=out/stage
    assert not list(plan.glob('applied_*'))
    root=Path(__file__).resolve().parents[1];geometry=root/'.Codex/r52_airlift_draft/ACTUAL_SERVICE_DOWNWARD_HATCH_MEASUREMENT_ROOT_FRAME_MINUS678_99.json'
    g=json.loads(geometry.read_text('utf8'));assert g['all_static_and_moving_extents_local'][3]-g['all_static_and_moving_extents_local'][0]<42
    lo=(-34,-750,307);hi=(95,-678,350);before,names=measure(lo,hi)
    for s in ('minecraft:air','projectseele:nerv_structural_panel','projectseele:nerv_shaft_panel','projectseele:nerv_machine_edge'):
        if s not in names:names.append(s)
    ids={s:i for i,s in enumerate(names)};after=before.copy()
    def fill(box,state):
        x,y,z,X,Y,Z=box;assert lo[0]<=x<=X<=hi[0]and lo[1]<=y<=Y<=hi[1]and lo[2]<=z<=Z<=hi[2],box
        after[y-lo[1]:Y-lo[1]+1,z-lo[2]:Z-lo[2]+1,x-lo[0]:X-lo[0]+1]=ids[state]
    gates=[]
    for variant,cx in enumerate((72,30,-12)):
        frame=[cx+.5,-678.99,328.5]
        fill((cx-18,-749,310,cx+18,-680,346),'projectseele:nerv_shaft_panel')
        fill((cx-16,-746,312,cx+16,-680,344),'minecraft:air')
        # Plate cassettes withdraw below the original sealed floor. They do
        # not cut another hole through the LCL perimeter or neighbouring bay.
        fill((cx-21,-685,310,cx+21,-680,346),'minecraft:air')
        # Short concrete corbels transfer cassette loads into the pit walls.
        # Full-height outer columns would cut the retained -705 escalator;
        # these stay above that original personnel corridor.
        for z in (311,345):fill((cx-21,-687,z,cx+21,-686,z),'projectseele:nerv_structural_panel')
        supports=[]
        for local in g['mandatory_bearing_points_local']:
            xyz=np.asarray(frame)+local;assert abs(xyz[1]+681)<1e-7
            x,z=math.floor(xyz[0]),math.floor(xyz[2]);fill((x,-685,z,x,-682,z),'projectseele:nerv_structural_panel')
            supports.append(dict(position=[x,-682,z],state='projectseele:nerv_structural_panel',source_contact=xyz.tolist()))
        cells=[]
        for x in range(cx-16,cx+17):
            for z in range(312,345):
                s=names[int(before[-679-lo[1],z-lo[2],x-lo[0]])]
                assert s.startswith('projectseele:nerv_')or s in('minecraft:iron_block','minecraft:light_gray_concrete','minecraft:polished_basalt[axis=y]','minecraft:sea_lantern','minecraft:polished_blackstone','minecraft:lodestone'),(variant,x,z,s)
                cells.append(dict(position=[x,-679,z],closed_state=s,open_state='minecraft:air'))
        gates.append(dict(id=f'bay_{variant}_service_pit',variant=variant,kind='SILO',frame=frame,
                          visual_uuid=str(uuid.uuid5(uuid.NAMESPACE_URL,f'projectseele:r52:bay_{variant}_service_pit')),
                          silo_hatch=dict(opening_x=33,opening_z=33,open_ticks=120,close_ticks=180,separation_sign=-1,beam_bottom=-2.01),
                          support_cells=supports,collision_cells=cells,clear_pit=[[cx-16,-746,312],[cx+16,-680,344]]))
    # Wet geometry and the full original solid floor stay exactly unchanged;
    # the runtime may open only after it actually drains all 90947 LCL cells.
    assert np.array_equal(after[-679-lo[1]:],before[-679-lo[1]:])
    tags=dict(iter_block_entities(BASE/'WORLD_R52','projectseele:geofront',lo,hi))
    for x,y,z in tags:assert before[y-lo[1],z-lo[2],x-lo[0]]==after[y-lo[1],z-lo[2],x-lo[0]]
    changed=before!=after;touched={names[int(v)]for v in np.unique(before[changed])}
    retained_mtr=np.isin(before,[i for i,s in enumerate(names)if s.startswith('mtr:')]);assert not np.any(changed&retained_mtr)
    assert all(s in ('minecraft:air','minecraft:stone','minecraft:deepslate','minecraft:deepslate[axis=y]','minecraft:bedrock','minecraft:sea_lantern')or s.startswith('projectseele:nerv_')for s in touched),touched
    vox.WORLD=BASE/'WORLD_R52';vox.OUT=out;vox.DIM='projectseele:geofront';p=vox.Painter(vertical_bounds=(-960,320))
    n=runs(p,lo,before,after,names,'R52_dry_mechanical_service_pit');p.meta.update(root_author=True,wet_floor_preserved=True,playable=False)
    plan=p.save_plan(stage)
    for gate in gates:
        q=plan/(gate['id']+'_collision.json.gz')
        with gzip.open(q,'wt',encoding='utf8')as f:json.dump(gate.pop('collision_cells'),f,separators=(',',':'))
        gate['collision_cells']={'file':q.name,'sha256':hashlib.sha256(q.read_bytes()).hexdigest(),'count':1089}
    report=dict(world_written=False,changed_cells=n,original_wet_floor_exact=True,gates=gates,original_devices_untouched=len(tags),below_pit_to_terminal_roof_metres=47,source_measurement_sha256=hashlib.sha256(geometry.read_bytes()).hexdigest(),pending=['real lower-frame/clamp sequencing','runtime watertight door commissioning','full original wet service native cycle'])
    (plan/'PLAN.json').write_text(json.dumps(report,indent=2),'utf8');np.savez_compressed(plan/'SECTION_Z328.npz',before=before[:,328-lo[2]],after=after[:,328-lo[2]],origin=lo,palette=names)
    print(json.dumps({k:v for k,v in report.items()if k!='gates'}),flush=True)
    if args.apply:
        receipt=p.apply(stage);actual,pal=measure(lo,hi)
        assert np.array_equal(np.asarray([ids[s]for s in pal],np.uint16)[actual],after)
        report.update(world_written=True,receipt=receipt,whole_pits_exact_readback=True);(plan/'PLAN.json').write_text(json.dumps(report,indent=2),'utf8')


if __name__=='__main__':main()
