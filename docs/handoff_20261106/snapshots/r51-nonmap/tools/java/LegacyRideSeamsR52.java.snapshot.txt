package com.projectseele.world;
import com.google.gson.*;
import java.nio.file.*;
import java.util.*;
import net.minecraft.world.phys.Vec3;
public final class LegacyRideSeamsR52 {
 private static double smooth(double t){return t*t*t*(t*(t*6-15)+10);}
 public static void main(String[] args)throws Exception {
  var d=NervRouteDirectoryR52.parse(JsonParser.parseString(Files.readString(Path.of(args[0]))).getAsJsonObject());var b=NervOriginalCarrierR52.nominalBinding(0);var l=d.leg("unit0/original_incline");var rows=new ArrayList<Object>();
  double[] centres={175.85506,190.95599,205.0511};
  for(double age:centres)for(int k=-3;k<=3;k++) {double time=age+k*.00001;var r=d.sample(l,smooth(time/240),b).evaRoot();var pose=NervLegacyCarrierRideR52.targetFromActualHeadsR52(r);var heads=new ArrayList<Object>();int i=0;
   for(int side:new int[]{-1,1})for(int row:new int[]{-1,1}){var p=r.transformPoint(new Vec3(side*6,-2.12,row*10));var h=NervLegacyCarrierRideR52.actualHeadR52(p.x,p.z);heads.add(Map.of("joint",i++,"z",p.z,"surface",h,"pose",pose));}
   rows.add(Map.of("age",time,"root",r,"heads",heads));
  }
  Files.writeString(Path.of(args[1]),new GsonBuilder().setPrettyPrinting().create().toJson(rows));System.out.println("seam rows "+rows.size());
 }
}
