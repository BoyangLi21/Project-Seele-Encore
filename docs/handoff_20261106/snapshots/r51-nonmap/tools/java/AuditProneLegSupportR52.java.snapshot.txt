    private static int checks;
    private static void check(boolean ok,String why){checks++;if(!ok)throw new AssertionError(why);}
    private static Vector3f v(JsonArray a){return new Vector3f(a.get(0).getAsFloat(),a.get(1).getAsFloat(),a.get(2).getAsFloat());}
    private static JsonObject json(Path p)throws Exception{return JsonParser.parseString(Files.readString(p)).getAsJsonObject();}
    private static JsonArray array(Vector3f p){var a=new JsonArray();a.add(p.x);a.add(p.y);a.add(p.z);return a;}
    private static EvaBodyPose.Sample rig(JsonObject data,int variant)
    {
        var bones=new LinkedHashMap<String,EvaBodyPose.Bone>();
        for(var raw:data.getAsJsonObject("rigs").getAsJsonArray(""+variant))
        {var b=raw.getAsJsonObject();var e=b.has("rotation")?v(b.getAsJsonArray("rotation")):b.has("bindRotationDegrees")?v(b.getAsJsonArray("bindRotationDegrees")):new Vector3f();e.mul((float)Math.PI/180);String n=b.get("name").getAsString();bones.put(n,new EvaBodyPose.Bone(n,b.has("parent")?b.get("parent").getAsString():null,v(b.getAsJsonArray("pivot")).mul(-1,1,1).div(16),new Quaternionf().rotationZYX(e.z,-e.y,-e.x)));}
        return new EvaBodyPose.Sample(Map.copyOf(bones));
    }
    private static EvaBodyPose.Sample clip(JsonObject data,int variant,String name,float phase)
    {
        var result=rig(data,variant);var capture=data.getAsJsonObject("stance_clips_by_rig").getAsJsonObject(""+variant);var names=capture.getAsJsonArray("bones");var frames=capture.getAsJsonObject("clips").getAsJsonObject(name).getAsJsonArray("frames");float at=phase*(frames.size()-1);int a=(int)at,b=Math.min(a+1,frames.size()-1);
        for(int i=0;i<names.size();i++)
        {String n=names.get(i).getAsString();var qs=new Quaternionf[2];var ps=new Vector3f[2];for(int k=0;k<2;k++){var f=frames.get(k==0?a:b).getAsJsonObject();var q=f.getAsJsonArray("rotation_wxyz").get(i).getAsJsonArray();qs[k]=new Quaternionf(-q.get(1).getAsFloat(),-q.get(2).getAsFloat(),q.get(3).getAsFloat(),q.get(0).getAsFloat()).normalize();ps[k]=n.equals("root")?v(f.getAsJsonArray("root_m")).mul(112):f.getAsJsonObject("bone_position_xyz").has(n)?v(f.getAsJsonObject("bone_position_xyz").getAsJsonArray(n)):new Vector3f();ps[k].mul(-1,1,1).div(16);}result.rotations.put(n,qs[0].slerp(qs[1],at-a));result.positions.put(n,ps[0].lerp(ps[1],at-a));}
        for(var bone:result.rig.values())if(bone.name().contains("_axis_"))result.rotations.put(bone.name(),new Quaternionf(bone.bindRotation()));
        EvaBodyPose.preserveJointCentres(result);return result;
    }
    private static EvaBodyPose.Sample nativePose(JsonObject capture,JsonObject extra)
    {
        var bones=new LinkedHashMap<String,EvaBodyPose.Bone>();var matrices=new HashMap<String,Matrix4f>();
        for(var e:extra.getAsJsonObject("actual_local_bones").entrySet()){var b=e.getValue().getAsJsonObject();bones.put(e.getKey(),new EvaBodyPose.Bone(e.getKey(),b.has("parent")?b.get("parent").getAsString():null,v(b.getAsJsonArray("pivot")),new Quaternionf()));}
        for(var e:capture.getAsJsonObject("bones").entrySet()){var a=e.getValue().getAsJsonObject().getAsJsonArray("matrix");var values=new float[16];for(int i=0;i<16;i++)values[i]=a.get(i).getAsFloat();matrices.put(e.getKey(),new Matrix4f().set(values));}
        var result=new EvaBodyPose.Sample(Map.copyOf(bones));
        for(var b:bones.values()){var local=new Matrix4f(matrices.get(b.name()));if(b.parent()!=null)local=new Matrix4f(matrices.get(b.parent())).invert().mul(local);var q=local.getUnnormalizedRotation(new Quaternionf()).normalize();result.rotations.put(b.name(),q);result.positions.put(b.name(),local.getTranslation(new Vector3f()).sub(b.pivot()).add(q.transform(new Vector3f(b.pivot()))));}result.dirty();
        for(var e:matrices.entrySet())check(difference(result.matrix(e.getKey()),e.getValue())<2e-5,"Same-frame native reconstruction failed: "+e.getKey());return result;
    }
    private static float difference(Matrix4f a,Matrix4f b){float error=0;var x=a.get(new float[16]);var y=b.get(new float[16]);for(int i=0;i<16;i++)error=java.lang.Math.max(error,java.lang.Math.abs(x[i]-y[i]));return error;}
    private static EvaAnatomicalHandsR45.Pose hands(ZipFile jar,int variant)throws Exception
    {
        var geo=JsonParser.parseString(new String(jar.getInputStream(jar.getEntry("assets/projectseele/geo/eva_unit0"+variant+".geo.json")).readAllBytes(),java.nio.charset.StandardCharsets.UTF_8)).getAsJsonObject();var rotations=new HashMap<String,Quaternionf>();
        for(var raw:geo.getAsJsonArray("minecraft:geometry").get(0).getAsJsonObject().getAsJsonArray("bones")){var b=raw.getAsJsonObject();String n=b.get("name").getAsString();if(!n.startsWith("r45_hand_"))continue;var e=v(b.getAsJsonArray("rotation")).mul((float)Math.PI/180);rotations.put(n,new Quaternionf().rotationZYX(e.z,-e.y,-e.x));}
        check(rotations.size()==34,"Complete original hand palette absent");return new EvaAnatomicalHandsR45.Pose("source","source",Map.copyOf(rotations));
    }
    private static JsonObject verify(String name,int variant,EvaBodyPose.Sample before,float stance,Vector3f terrain,EvaOriginalMeshGeometryR52.Provider provider,EvaAnatomicalHandsR45.Pose hands,Path out,boolean cloud)throws Exception
    {
        var pose=EvaBodyPose.copy(before);EvaProneLegSupportR52.apply(variant,pose,stance,terrain);var row=new JsonObject();row.addProperty("case",name);row.addProperty("rig",variant);row.addProperty("stance",stance);var lows=new JsonObject();var pivots=new JsonObject();var beforeLows=new JsonObject();for(var entry:EvaProneLegSupportR52.sourceVerticesR52(variant).entrySet()){float low=Float.POSITIVE_INFINITY;var matrix=before.matrix(entry.getKey());for(var vertex:entry.getValue())low=Math.min(low,matrix.transformPosition(new Vector3f(vertex)).y*5);beforeLows.addProperty(entry.getKey(),low);}row.add("before_skin_minimum_m",beforeLows);
        for(String side:List.of("l","r"))
        {
            String leg="leg_"+side,shin="shin_"+side,foot="foot_"+side;
            check(difference(before.matrix(foot),pose.matrix(foot))<4e-5,"Complete original shoe world matrix changed: "+name+side);
            check(before.rotations.get(shin).equals(pose.rotations.get(shin),0),"Original knee hinge changed");check(before.positions.get(shin).equals(pose.positions.get(shin)),"Original joint centre offset changed");
            for(String n:List.of(leg,foot)){var a=before.matrix(n).transformPosition(new Vector3f(before.rig.get(n).pivot()));var b=pose.matrix(n).transformPosition(new Vector3f(pose.rig.get(n).pivot()));check(a.distance(b)<3e-5,"Fixed hip/ankle centre moved");pivots.add(n,array(b.mul(5)));}
            float angle=2*(float)java.lang.Math.acos(java.lang.Math.min(1,java.lang.Math.abs(before.rotations.get(leg).dot(pose.rotations.get(leg)))));check(angle<=java.lang.Math.toRadians(75.001),"Excessive chain swivel");row.addProperty("swivel_"+side+"_degrees",java.lang.Math.toDegrees(angle));
        }
        for(String n:List.of("root","torso_lower","torso_upper","head","arm_l","arm_r","hand_l","hand_r"))check(difference(before.matrix(n),pose.matrix(n))<1e-6,"Unrelated production pose changed: "+n);
        var geometry=provider.evaluate(variant,pose,hands);var normal=new Quaternionf().rotationX(terrain.x).rotateZ(terrain.y).transform(new Vector3f(0,1,0));float plane=normal.y*terrain.z;
        for(var part:geometry.parts())if(part.bone().startsWith("leg_")||part.bone().startsWith("shin_")||part.bone().startsWith("foot_"))
        {double low=Double.POSITIVE_INFINITY;var t=part.triangles();for(int i=0;i<t.length;i+=3)low=java.lang.Math.min(low,normal.x*t[i]+normal.y*t[i+1]+normal.z*t[i+2]-plane);lows.addProperty(part.bone(),low);if(stance==3&&!part.bone().startsWith("foot_"))check(low>=.0298,"Complete original skin remains below support: "+name+" "+part.bone()+" "+low);}
        row.add("full_original_skin_minimum_m",lows);row.add("fixed_joint_centres_m",pivots);
        if(cloud)
        {
            var originalGeometry=provider.evaluate(variant,before,hands);var contactErrors=new JsonObject();
            for(String side:List.of("l","r")){String n="foot_"+side;var a=originalGeometry.parts().stream().filter(p->p.bone().equals(n)).findFirst().orElseThrow().triangles();var b=geometry.parts().stream().filter(p->p.bone().equals(n)).findFirst().orElseThrow().triangles();int low=0;for(int i=3;i<a.length;i+=3)if(a[i+1]<a[low+1])low=i;double error=Math.sqrt(Math.pow(a[low]-b[low],2)+Math.pow(a[low+1]-b[low+1],2)+Math.pow(a[low+2]-b[low+2],2));check(error<.0005,"True original source shoe contact vertex moved: "+name+n);contactErrors.addProperty(side,error);}row.add("original_actual_lowest_shoe_vertex_error_m",contactErrors);
            var meshes=new HashMap<String,NervRigidMeshSweepR52>();for(var part:geometry.parts())if(part.bone().startsWith("leg_")||part.bone().startsWith("shin_")||part.bone().startsWith("foot_"))meshes.put(part.bone(),new NervRigidMeshSweepR52(part.triangles()));
            var frame=RigidTransform.of(net.minecraft.world.phys.Vec3.ZERO,new Quaternionf());var adjacency=new JsonObject();
            for(String a:List.of("leg_l","shin_l","foot_l"))for(String b:List.of("leg_r","shin_r","foot_r"))check(!meshes.get(a).intersects(frame,frame,meshes.get(b),frame,frame,.001,net.minecraft.world.phys.Vec3.ZERO),"Opposite complete leg skins intersect: "+name+" "+a+":"+b);
            for(String side:List.of("l","r"))adjacency.addProperty("shin_foot_surface_contact_"+side,meshes.get("shin_"+side).intersects(frame,frame,meshes.get("foot_"+side),frame,frame,.001,net.minecraft.world.phys.Vec3.ZERO));
            row.add("actual_source_ankle_interface",adjacency);
        }
        if(cloud){var matrices=new JsonObject();for(String n:pose.rig.keySet()){var a=new JsonArray();for(float f:pose.matrix(n).get(new float[16]))a.add(f);matrices.add(n,a);}Files.writeString(out.resolve(name+"_matrices.json"),new GsonBuilder().setPrettyPrinting().create().toJson(matrices));}
        if(stance<=1)for(String n:pose.rig.keySet())check(difference(before.matrix(n),pose.matrix(n))==0,"Standing/crouching pose changed");
        if(cloud)try(var stream=new DataOutputStream(new GZIPOutputStream(Files.newOutputStream(out.resolve(name+".bin.gz")))))
        {stream.writeInt(0x52353243);stream.writeInt(1);stream.writeInt(geometry.parts().size());for(var part:geometry.parts()){stream.writeUTF(part.bone());stream.writeUTF(part.sourceSHA256());stream.writeInt(part.triangles().length/3);for(double d:part.triangles())stream.writeFloat((float)d);}}
        return row;
    }
    private static JsonObject benchmark(int variant,EvaBodyPose.Sample pose)
    {
        var saved=new HashMap<String,Quaternionf>();for(String n:List.of("leg_l","leg_r","foot_l","foot_r"))saved.put(n,new Quaternionf(pose.rotations.get(n)));long[] times=new long[5];
        for(int batch=-1;batch<5;batch++)
        {
            int samples=batch<0?3000:5000;long start=System.nanoTime();
            for(int at=0;at<samples;at++){saved.forEach((n,q)->pose.rotations.get(n).set(q));pose.dirty();EvaProneLegSupportR52.apply(variant,pose,3,new Vector3f());}
            if(batch>=0)times[batch]=(System.nanoTime()-start)/samples;
        }
        Arrays.sort(times);var row=new JsonObject();row.addProperty("rig",variant);row.addProperty("cached_support_hull_points",EvaProneLegSupportR52.sourceVerticesR52(variant).values().stream().mapToInt(a->a.length).sum());row.addProperty("warmup_samples",3000);row.addProperty("timed_samples",25000);row.addProperty("median_ms_per_sample",times[2]/1e6);row.addProperty("fastest_batch_ms_per_sample",times[0]/1e6);row.addProperty("slowest_batch_ms_per_sample",times[4]/1e6);row.addProperty("scope","Warm production helper plus lower-channel reset/FK invalidation; this host CPU only, not native frame-rate acceptance");return row;
    }
    public static void main(String[] args)throws Exception
    {
        Path out=Path.of(args[4]);Files.createDirectories(out);var data=json(Path.of(args[1]));var physical=json(Path.of(args[2]));var capture=json(Path.of(args[3]).resolve("native_prone_cannon.json"));var extra=json(Path.of(args[3]).resolve("native_prone_cannon_extra.json"));
        check(capture.get("powered").getAsBoolean()&&!capture.get("shutdown").getAsBoolean()&&capture.get("server_stable_ticks").getAsInt()>=20&&capture.get("stance").getAsInt()==3,"Actual powered stable prone required");
        var rows=new JsonArray();
        try(var jar=new ZipFile(args[0]))
        {
            var receipt=JsonParser.parseString(new String(jar.getInputStream(jar.getEntry("assets/projectseele/eva/eva_original_mesh_geometry_r52_sources.json")).readAllBytes(),java.nio.charset.StandardCharsets.UTF_8)).getAsJsonObject();var hashes=new HashMap<String,String>();for(var e:receipt.getAsJsonObject("resources").entrySet())hashes.put(e.getKey(),e.getValue().getAsString());
            var provider=EvaOriginalMeshGeometryR52.load(r->{var e=jar.getEntry("assets/projectseele/"+r);return jar.getInputStream(e).readAllBytes();},hashes);
            var nativeFingerRotations=new HashMap<String,Quaternionf>();for(var e:capture.getAsJsonObject("actual_finger_quaternions").entrySet()){var q=e.getValue().getAsJsonArray();nativeFingerRotations.put(e.getKey(),new Quaternionf(q.get(0).getAsFloat(),q.get(1).getAsFloat(),q.get(2).getAsFloat(),q.get(3).getAsFloat()).normalize());}check(nativeFingerRotations.size()==34,"Same-frame actual native finger palette required");var nativeHands=new EvaAnatomicalHandsR45.Pose("actual_native_cannon","actual_native_cannon",Map.copyOf(nativeFingerRotations));rows.add(verify("powered_native_01",1,nativePose(capture,extra),3,new Vector3f(),provider,nativeHands,out,true));
            for(int variant=0;variant<3;variant++)for(String name:List.of("unarmed_stance","rifle_stance"))for(float stance:new float[]{0,1,1.5F,2,2.05F,2.1F,2.2F,2.3F,2.4F,2.5F,2.6F,2.8F,3})
            {
                var pose=clip(data,variant,name,stance/3);var profile=new CombatBodyProfiles.Profile(""+variant,physical.getAsJsonObject("models").getAsJsonObject(""+variant),pose.rig,null);AnatomicalLimbConstraints.apply(pose,profile);
                rows.add(verify("original_0"+variant+"_"+name+"_"+stance,variant,pose,stance,new Vector3f(),provider,hands(jar,variant),out,stance==3));
            }
            for(int variant=0;variant<3;variant++)for(int step=0;step<=12;step++)for(float moving:new float[]{.5F,1})
            {
                var crawl=clip(data,variant,"prone_crawl",step/12F);var pose=moving==1?crawl:EvaBodyPose.blend(clip(data,variant,"unarmed_stance",1),crawl,moving);var profile=new CombatBodyProfiles.Profile(""+variant,physical.getAsJsonObject("models").getAsJsonObject(""+variant),pose.rig,null);AnatomicalLimbConstraints.apply(pose,profile);rows.add(verify("original_0"+variant+"_prone_crawl_phase_"+step+"_weight_"+moving,variant,pose,3,new Vector3f(),provider,hands(jar,variant),out,false));
            }
            for(int variant=0;variant<3;variant++)for(String name:List.of("unarmed_stance","rifle_stance"))for(int sign:new int[]{-1,1})
            {
                var pose=clip(data,variant,name,1);var profile=new CombatBodyProfiles.Profile(""+variant,physical.getAsJsonObject("models").getAsJsonObject(""+variant),pose.rig,null);AnatomicalLimbConstraints.apply(pose,profile);var terrain=new Vector3f(sign*.1F,-sign*.1F,sign*1.2F);pose.rotations.get("root").rotateX(terrain.x).rotateZ(terrain.y);pose.positions.get("root").y+=terrain.z/5;pose.dirty();rows.add(verify("original_0"+variant+"_"+name+"_support_plane_"+sign,variant,pose,3,terrain,provider,hands(jar,variant),out,false));
            }
            var continuity=new JsonArray();
            for(int variant=0;variant<3;variant++)for(String name:List.of("unarmed_stance","rifle_stance"))
            {
                EvaBodyPose.Sample previous=null;float greatestStep=0,greatestStance=0;int fullVertexClearSamples=0;
                for(int step=0;step<=200;step++)
                {
                    float stance=1+step/100F;var pose=clip(data,variant,name,stance/3);var profile=new CombatBodyProfiles.Profile(""+variant,physical.getAsJsonObject("models").getAsJsonObject(""+variant),pose.rig,null);AnatomicalLimbConstraints.apply(pose,profile);var before=EvaBodyPose.copy(pose);EvaProneLegSupportR52.apply(variant,pose,stance,new Vector3f());
                    for(String side:List.of("l","r"))
                    {
                        check(difference(before.matrix("foot_"+side),pose.matrix("foot_"+side))<4e-5,"Dense transition displaced original shoe");
                        if(previous!=null){var a=previous.matrix("leg_"+side).getUnnormalizedRotation(new Quaternionf()).normalize();var b=pose.matrix("leg_"+side).getUnnormalizedRotation(new Quaternionf()).normalize();float angle=2*(float)Math.acos(Math.min(1,Math.abs(a.dot(b))));if(angle>greatestStep){greatestStep=angle;greatestStance=stance;}}
                    }
                    float low=Float.POSITIVE_INFINITY;for(var e:EvaProneLegSupportR52.sourceVerticesR52(variant).entrySet()){var m=pose.matrix(e.getKey());for(var vertex:e.getValue())low=Math.min(low,m.transformPosition(new Vector3f(vertex)).y);}

                }
                check(greatestStep<Math.toRadians(8),"Prone support introduced a transition jump: "+variant+" "+name+" at "+greatestStance+" delta "+Math.toDegrees(greatestStep));var r=new JsonObject();r.addProperty("rig",variant);r.addProperty("clip",name);r.addProperty("original_stance_samples",fullVertexClearSamples);r.addProperty("greatest_leg_world_step_degrees_per_0_01_stance",Math.toDegrees(greatestStep));continuity.add(r);
            }
            var crawlCycles=new JsonArray();
            for(int variant=0;variant<3;variant++)for(float moving:new float[]{.5F,1})
            {
                float least=Float.POSITIVE_INFINITY,greatest=0;
                for(int step=0;step<=200;step++)

                    float low=Float.POSITIVE_INFINITY;for(var e:EvaProneLegSupportR52.sourceVerticesR52(variant).entrySet()){var m=pose.matrix(e.getKey());for(var vertex:e.getValue())low=Math.min(low,m.transformPosition(new Vector3f(vertex)).y);}

                }
                var r=new JsonObject();r.addProperty("rig",variant);r.addProperty("move_weight",moving);r.addProperty("cycle_samples",201);r.addProperty("complete_leg_skin_minimum_m",least);r.addProperty("largest_swivel_degrees",Math.toDegrees(greatest));crawlCycles.add(r);

            var report=new JsonObject();report.addProperty("scope","Unchanged selected full original mesh and original R48 stance source; powered native 01 same-frame FK; source/pose audit only, not new native station acceptance");report.addProperty("checks",checks);report.addProperty("source",provider.sourceFingerprint());report.add("cases",rows);report.add("dense_original_prone_transitions",continuity);report.add("dense_original_crawl_cycles",crawlCycles);report.add("warm_production_helper_cpu_microbenchmark",performance);Files.writeString(out.resolve("PRONE_LEG_SOURCE_AUDIT.json"),new GsonBuilder().setPrettyPrinting().create().toJson(report));
        }
        System.out.println("Original prone source audit passed: "+checks+" checks, "+rows.size()+" source cases");
    }
