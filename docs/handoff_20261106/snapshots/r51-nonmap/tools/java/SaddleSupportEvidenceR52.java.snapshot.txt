import com.google.gson.*;
import java.nio.file.*;
import java.util.*;
import com.projectseele.world.NervFootGuideDockR52.*;
public final class SaddleSupportEvidenceR52 {
 public static void main(String[] args)throws Exception{var d=NervRouteDirectoryR52.parse(JsonParser.parseString(Files.readString(Path.of(args[0]))).getAsJsonObject());var stations=JsonParser.parseString(Files.readString(Path.of(args[1]))).getAsJsonObject().getAsJsonArray("stations");var rows=new ArrayList<Object>();int failed=0;for(var raw:stations){var r=raw.getAsJsonObject();String id=r.get("leg").getAsString();var spec=NervMechanicalRailBindingR52.stationSpec(d,new NervMechanicalRailBindingR52.Spec(NervMechanicalRailBindingR52.Kind.FOOT_TURN_STATION,id,NervFootGuideDockR52.SOURCE_SHA,NervFootGuideRigR52.SOURCE_SHA,NervFootGuideStationR52.proposedIdentityR52(d.epoch(),id),r.get("foundation_top").getAsDouble()));var view=NervFootGuideDockR52.view(spec);double maximum=0,minArea=1e99,minNormal=1;int count=0;var failures=new ArrayList<Object>();for(int i=0;i<=16;i++)for(var pose:List.of(new StationPose(i/16D,1,1,0,1),new StationPose(i/16D,1,1,0,0),new StationPose(0,1,1,0,i/16D),new StationPose(1,1,1,0,i/16D))){for(var proof:NervFootGuideDockR52.saddleSupportsR52(view,pose)){count++;maximum=Math.max(maximum,proof.maximumFit());minArea=Math.min(minArea,proof.finiteArea());minNormal=Math.min(minNormal,proof.minimumNormalAgreement());if(!proof.bears()){failed++;if(failures.size()<10)failures.add(Map.of("pose",pose,"actual",proof));}}}rows.add(Map.of("leg",id,"proofs",count,"maximum_fit_m",maximum,"minimum_finite_area_m2",minArea,"minimum_normal_agreement",minNormal,"first_failures",failures));System.out.println(id+" max="+maximum+" area="+minArea+" failures="+failures.size());}Files.writeString(Path.of(args[2]),new GsonBuilder().setPrettyPrinting().create().toJson(Map.of("world_written",false,"native_accepted",false,"source",NervFootGuideDockR52.class.getField("SOURCE_SHA").get(null),"total_failed",failed,"rows",rows)));}
}
