package com.projectseele.world;
import com.google.gson.*;
import java.nio.file.*;
import java.util.*;
import net.minecraft.world.phys.*;
/** Whole selected goal EVA/hands and original carrier on the unchanged
 * original R20 and bank lines, paired against every legacy steel part. */
public final class LegacyFullAssemblyEvidenceR52 {
 private record Body(String name,NervRigidMeshSweepR52 mesh){}
 public static void main(String[] args)throws Exception{var d=NervRouteDirectoryR52.parse(JsonParser.parseString(Files.readString(Path.of(args[0]))).getAsJsonObject());JsonObject goals;try(var stream=new java.util.zip.GZIPInputStream(Files.newInputStream(Path.of(args[1])))){goals=JsonParser.parseReader(new java.io.InputStreamReader(stream,java.nio.charset.StandardCharsets.UTF_8)).getAsJsonObject();}var fixed=NervLegacyCarrierRideR52.fixedPartsR52();var boxes=new ArrayList<AABB>();for(var p:fixed)boxes.add(p.mesh().envelope(p.frame(),p.frame()));var rows=new ArrayList<Object>();
  for(int v=0;v<3;v++){var body=new ArrayList<Body>();for(var raw:goals.getAsJsonArray("variants")){var g=raw.getAsJsonObject();if(g.get("variant").getAsInt()!=v)continue;for(var r:g.getAsJsonArray("parts")){var p=r.getAsJsonObject();var a=p.getAsJsonArray("triangles");double[] xyz=new double[a.size()];for(int i=0;i<a.size();i++)xyz[i]=a.get(i).getAsDouble();body.add(new Body("EVA/"+p.get("bone").getAsString(),new NervRigidMeshSweepR52(xyz)));}}var binding=NervOriginalCarrierR52.nominalBinding(v);
   for(String suffix:List.of("original_incline","bank1_to_2","bank2_to_3")){String id="unit"+v+"/"+suffix;var leg=d.leg(id);var hits=new TreeSet<String>();int finite=0,missing=0,soleReceipts=0;double minimumSoleArea=1e99,minimumSoleExterior=1e99,maximumSoleGap=0;for(int k=0;k<=leg.ticks();k++){var root=d.sample(leg,NervMechanicalTravelR52.fraction(k,leg.ticks()),binding).evaRoot();var pose=NervLegacyCarrierRideR52.targetFromActualHeadsR52(root);if(pose==null){missing++;continue;}for(var proof:NervLegacyCarrierRideR52.finitePadBearingsR52(root,pose))if(proof.bears())finite++;var carrier=NervOriginalCarrierR52.partsAtR52(v,root,1,new Vec3(0,1,0),pose);
    for(int i=0;i<fixed.size();i++){var f=fixed.get(i);var fb=boxes.get(i);for(var p:carrier){if(!p.mesh().envelope(p.frame(),p.frame()).intersects(fb))continue;if(NervLegacyCarrierRideR52.namedHeadContactR52(p,p,f)||NervLegacyCarrierRideR52.namedDeckHeadContactR52(p,p,f))continue;var soleProofs=NervLegacyCarrierRideR52.finiteSoleHeadFacesR52(p,p,f);if(soleProofs.size()==2&&soleProofs.stream().allMatch(NervLegacyCarrierRideR52.SoleHeadBearing::bears)){soleReceipts++;for(var proof:soleProofs){minimumSoleArea=Math.min(minimumSoleArea,proof.finiteArea());minimumSoleExterior=Math.min(minimumSoleExterior,proof.minimumExterior());maximumSoleGap=Math.max(maximumSoleGap,proof.maximumFacetGap());}continue;}if(p.mesh().intersects(p.frame(),p.frame(),f.mesh(),f.frame(),f.frame(),.003,Vec3.ZERO))hits.add(p.name()+" vs "+f.name()+" tick="+k);}for(var p:body)if(p.mesh().envelope(root,root).intersects(fb)&&p.mesh().intersects(root,root,f.mesh(),f.frame(),f.frame(),.003,Vec3.ZERO))hits.add(p.name()+" vs "+f.name()+" tick="+k);}
   }rows.add(Map.of("leg",id,"samples",leg.ticks()+1,"target_missing",missing,"finite_shoe_contacts",finite,"expected_finite_contacts",4*(leg.ticks()+1),"source_hits",hits,"sole_facet_receipts",Map.of("accepted_pairs",soleReceipts,"minimum_actual_overlap_area_m2",minimumSoleArea,"minimum_whole_sole_exterior_m",minimumSoleExterior,"maximum_contact_facet_gap_m",maximumSoleGap)));System.out.println(id+" missing="+missing+" hits="+hits.size());Files.writeString(Path.of(args[2]),new GsonBuilder().setPrettyPrinting().create().toJson(Map.of("world_written",false,"native_accepted",false,"includes_full_goal_eva_and_original_carrier",true,"source",NervLegacyCarrierRideR52.class.getField("SOURCE_SHA").get(null),"full_source",NervOriginalCarrierR52.class.getField("SOURCE_SHA").get(null),"rows",rows)));}
  }
 }
}
