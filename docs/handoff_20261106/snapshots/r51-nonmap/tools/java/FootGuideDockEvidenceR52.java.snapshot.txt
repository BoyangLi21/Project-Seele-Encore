    private static final Map<Integer,Set<String>> extraHits=new TreeMap<>();

    public static void main(String[] args)throws Exception {
        if(args.length>4){var cells=JsonParser.parseString(Files.readString(Path.of(args[4]))).getAsJsonObject().getAsJsonArray("candidate_cells");for(var raw:cells){var a=raw.getAsJsonArray();extraCubes.add(new AABB(a.get(0).getAsDouble(),a.get(1).getAsDouble(),a.get(2).getAsDouble(),a.get(0).getAsDouble()+1,a.get(1).getAsDouble()+1,a.get(2).getAsDouble()+1));}}
        if(args.length>3)loadCandidates(Path.of(args[3]));
        EvaOriginalMeshGeometryR52.installClasspath();
        var directory=NervRouteDirectoryR52.parse(JsonParser.parseString(Files.readString(Path.of(args[0]))).getAsJsonObject());
        var bodyJson=JsonParser.parseString(Files.readString(Path.of(args[1]))).getAsJsonObject();var maker=EvaOriginalMeshGeometryR52Test.class.getDeclaredMethod("body",JsonObject.class,int.class);maker.setAccessible(true);var provider=EvaOriginalMeshGeometryR52.readOnlySelectedProviderR52();var units=new ArrayList<Map<String,Object>>();

            if(in==null||out==null)throw new AssertionError("Existing straight Root guide adjacency required: "+id);
            AABB assembly=null;for(int k=0;k<=32;k++){var root=directory.sample(leg,k/32D,binding).evaRoot();for(var part:NervOriginalCarrierR52.partsAtR52(v,root,1,new Vec3(0,1,0))){var b=part.mesh().envelope(part.frame(),part.frame());assembly=assembly==null?b:assembly.minmax(b);}for(var part:eva.parts()){var b=NervRigidMeshSweepR52.transformed(part.bounds(),root);assembly=assembly.minmax(b);}}
            double floor=Math.floor(assembly.minY-.5);var spec=new Spec(directory,leg,binding,in,out,floor);var footRows=new ArrayList<Map<String,Object>>();for(var f:NervFootGuideDockR52.feet(spec))footRows.add(Map.of("name",f.name(),"centre",new double[]{f.centre().x,f.centre().y,f.centre().z},"half_xz",new double[]{f.halfX(),f.halfZ()}));
            var evaMeshes=new LinkedHashMap<String,NervRigidMeshSweepR52>();for(var part:eva.parts())evaMeshes.put(part.bone()+"/"+evaMeshes.size(),new NervRigidMeshSweepR52(part.triangles()));
            var hits=new LinkedHashSet<String>();var wholeHits=new LinkedHashSet<String>();AABB hardware=null;int finite=0;var boreRows=new ArrayList<Object>();
            for(int k=0;k<=16;k++){
                var at=new Sample(Stage.TURN,k/16D);var root=NervFootGuideDockR52.root(spec,at);var parts=NervFootGuideDockR52.pieces(spec,at);if(NervFootGuideDockR52.finiteAxleContacts(spec,at))finite++;
                if(k%4==0)for(var f:NervFootGuideDockR52.bearingFaces(binding,root,NervFootGuideDockR52.guideUp(spec,at)))boreRows.add(Map.of("fraction",k/16D,"side",f.side(),"finite_faces",f.faces(),"area_m2",f.area(),"maximum_separation_m",f.maximumSeparation(),"maximum_compression_m",f.maximumCompression(),"minimum_normal_agreement",f.minimumNormalAgreement(),"bears",f.bears()));
                var carrier=candidates(NervOriginalCarrierR52.partsAtR52(v,root,1,NervFootGuideDockR52.guideUp(spec,at)));
                var guideDraws=NervFootGuideRigR52.draws(binding,root,1,NervFootGuideDockR52.guideUp(spec,at));if(!Boolean.getBoolean("projectseele.dock.stationOnly")&&!Boolean.getBoolean("projectseele.dock.extraOnly"))for(var fixed:NervFixedFootGuidesR52.pieces()){var fb=fixed.mesh().envelope(fixed.frame(),fixed.frame());for(var q:carrier){if(!q.mesh().envelope(q.frame(),q.frame()).intersects(fb))continue;var draw=guideDraws.stream().filter(g->q.name().equals("guide/"+g.part().name())).findFirst().orElse(null);if(draw!=null&&NervFootGuideRigR52.namedGuideContactR52(draw,draw,fixed))continue;if(q.mesh().intersects(q.frame(),q.frame(),fixed.mesh(),fixed.frame(),fixed.frame(),.003,Vec3.ZERO))wholeHits.add(q.name()+" vs "+fixed.name()+" @ "+k);}for(var q:evaMeshes.entrySet())if(NervRigidMeshSweepR52.transformed(q.getValue().bounds(),root).intersects(fb)&&q.getValue().intersects(root,root,fixed.mesh(),fixed.frame(),fixed.frame(),.003,Vec3.ZERO))wholeHits.add("EVA/"+q.getKey()+" vs "+fixed.name()+" @ "+k);}
                if(!Boolean.getBoolean("projectseele.dock.stationOnly")&&!Boolean.getBoolean("projectseele.dock.extraOnly"))for(var anchor:NervFixedFootGuidesR52.anchors()){var voxel=new AABB(anchor.at());for(var q:carrier)if(q.mesh().envelope(q.frame(),q.frame()).intersects(voxel)&&q.mesh().intersects(q.frame(),q.frame(),voxel,.003))wholeHits.add(q.name()+" vs future_1m_cube/"+anchor.path()+"/"+anchor.at()+" @ "+k);for(var q:evaMeshes.entrySet())if(NervRigidMeshSweepR52.transformed(q.getValue().bounds(),root).intersects(voxel)&&q.getValue().intersects(root,root,voxel,.003))wholeHits.add("EVA/"+q.getKey()+" vs future_1m_cube/"+anchor.path()+"/"+anchor.at()+" @ "+k);}
                AABB completeFrame=null;for(var q:carrier){var b=q.mesh().envelope(q.frame(),q.frame());completeFrame=completeFrame==null?b:completeFrame.minmax(b);}for(var q:evaMeshes.values())completeFrame=completeFrame.minmax(NervRigidMeshSweepR52.transformed(q.bounds(),root));for(var q:parts)completeFrame=completeFrame.minmax(q.mesh().envelope(q.frame(),q.frame()));
                for(int index=0;index<extraCubes.size();index++){var voxel=extraCubes.get(index);if(!completeFrame.intersects(voxel))continue;for(var q:carrier)if(q.mesh().envelope(q.frame(),q.frame()).intersects(voxel)&&q.mesh().intersects(q.frame(),q.frame(),voxel,.003))wholeHits.add(q.name()+" vs new_support_cube/"+index+" @ "+k);for(var q:evaMeshes.entrySet())if(NervRigidMeshSweepR52.transformed(q.getValue().bounds(),root).intersects(voxel)&&q.getValue().intersects(root,root,voxel,.003))wholeHits.add("EVA/"+q.getKey()+" vs new_support_cube/"+index+" @ "+k);for(var q:parts)if(q.mesh().envelope(q.frame(),q.frame()).intersects(voxel)&&q.mesh().intersects(q.frame(),q.frame(),voxel,.003))wholeHits.add("station/"+q.name()+" vs new_support_cube/"+index+" @ "+k);}
                if(!Boolean.getBoolean("projectseele.dock.extraOnly"))for(var p:parts){var b=p.mesh().envelope(p.frame(),p.frame());hardware=hardware==null?b:hardware.minmax(b);
                    var local=root.inverse().compose(p.frame());var localBounds=p.mesh().envelope(local,local);
                    if(!Boolean.getBoolean("projectseele.dock.stationOnly"))for(var q:carrier){var qf=root.inverse().compose(q.frame());if(!localBounds.intersects(q.mesh().envelope(qf,qf)))continue;var view=NervFootGuideDockR52.view(spec);var pose=NervFootGuideDockR52.stationPose(spec,at);if(NervFootGuideDockR52.namedStationContactR52(q,q,view,pose,pose,p))continue;if(p.mesh().intersects(p.frame(),p.frame(),q.mesh(),q.frame(),q.frame(),.001,Vec3.ZERO))hits.add(p.name()+" vs "+q.name());}
                    if(!Boolean.getBoolean("projectseele.dock.stationOnly"))for(var q:evaMeshes.entrySet())if(localBounds.intersects(q.getValue().bounds())&&p.mesh().intersects(p.frame(),p.frame(),q.getValue(),root,root,.001,Vec3.ZERO))hits.add(p.name()+" vs EVA/"+q.getKey());
                    for(var fixed:NervFixedFootGuidesR52.pieces())if(b.intersects(fixed.mesh().envelope(fixed.frame(),fixed.frame()))&&!NervFootGuideDockR52.namedFixedEndContactR52(p,p,fixed)&&p.mesh().intersects(p.frame(),p.frame(),fixed.mesh(),fixed.frame(),fixed.frame(),.003,Vec3.ZERO))hits.add(p.name()+" vs "+fixed.name());
                }
            }
            for(String hit:wholeHits){int marker=hit.indexOf("new_support_cube/");if(marker>=0){int end=hit.indexOf(" @",marker);int index=Integer.parseInt(hit.substring(marker+17,end));extraHits.computeIfAbsent(index,key->new LinkedHashSet<>()).add(id+" "+hit);}}
            var row=new LinkedHashMap<String,Object>();row.put("leg",id);row.put("reverse_leg",id+"/reverse");row.put("variant",v);row.put("whole_assembly_bounds",bounds(assembly));row.put("foundation_top",floor);row.put("hardware_bounds",bounds(hardware));row.put("foundation_feet",footRows);row.put("actual_source_axle_finite_faces_pass",finite);row.put("sample_count",17);row.put("source_hits",hits);row.put("whole_eva_carrier_vs_fixed_and_1m_cubes",wholeHits);row.put("actual_lower_bore_faces",boreRows);row.put("extra_support_only",Boolean.getBoolean("projectseele.dock.extraOnly"));row.put("extra_cube_count",extraCubes.size());row.put("station_only",Boolean.getBoolean("projectseele.dock.stationOnly"));row.put("contact_scope","3 noncollinear real cap face probes per side, .005m2; Root <=0.2mm named finite bearing compression only");row.put("from_up",new double[]{in.x,in.y,in.z});row.put("to_up",new double[]{out.x,out.y,out.z});units.add(row);System.out.println(id+" finite="+finite+" station_hits="+hits.size()+" whole_hits="+wholeHits.size()+" floor="+floor);Files.writeString(Path.of(args[2]),new GsonBuilder().setPrettyPrinting().create().toJson(Map.of("world_written",false,"native_admission",false,"station_source",NervFootGuideDockR52.SOURCE_SHA,"fixed_source",NervFixedFootGuidesR52.SOURCE_SHA,"candidate_stations",units)));
        }

        var result=Map.of("candidate_support_cells",perCell,"world_written",false,"native_admission",false,"station_source",NervFootGuideDockR52.class.getField("SOURCE_SHA").get(null),"full_source",NervOriginalCarrierR52.class.getField("SOURCE_SHA").get(null),"original_guide_source",NervFootGuideRigR52.SOURCE_SHA,"candidate_stations",units);Path path=Path.of(args[2]);Files.createDirectories(path.getParent());Files.writeString(path,new GsonBuilder().setPrettyPrinting().create().toJson(result));
    }
