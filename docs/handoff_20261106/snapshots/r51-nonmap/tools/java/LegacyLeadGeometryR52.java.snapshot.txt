 private static RigidTransform frame(JsonObject j){var p=j.getAsJsonArray("position");var q=j.getAsJsonArray("quaternion");return RigidTransform.of(new Vec3(p.get(0).getAsDouble(),p.get(1).getAsDouble(),p.get(2).getAsDouble()),new Quaternionf(q.get(0).getAsFloat(),q.get(1).getAsFloat(),q.get(2).getAsFloat(),q.get(3).getAsFloat()));}
 private static final Map<String,NervRigidMeshSweepR52> candidates=new HashMap<>();
 private static void load(Path folder)throws Exception{for(String file:List.of("carrier_body_turn_slots_candidate_r52.json","carrier_deck_turn_slots_candidate_r52.json","carrier_service_stow_turn_candidate_r52.json")){var j=JsonParser.parseString(Files.readString(folder.resolve(file))).getAsJsonObject();int stride=j.get("stride").getAsInt();for(var entry:j.getAsJsonObject("parts").entrySet()){var a=entry.getValue().getAsJsonArray();double[] xyz=new double[a.size()/stride*3];for(int i=0,k=0;i<a.size();i+=stride){xyz[k++]=a.get(i).getAsDouble();xyz[k++]=a.get(i+1).getAsDouble();xyz[k++]=a.get(i+2).getAsDouble();}candidates.put(entry.getKey(),new NervRigidMeshSweepR52(xyz));}}}
 private static final Map<String,NervRigidMeshSweepR52> gantryMeshes=new HashMap<>();
 private static List<NervFlightProofR52.ObstacleMesh> gantry(int variant,RigidTransform frame,float closed,double lower)throws Exception{var j=JsonParser.parseString(new String(EvaOriginalMeshGeometryR52.classpathBytes("mesh/tv_shoulder_shells_r44.json"),java.nio.charset.StandardCharsets.UTF_8)).getAsJsonObject();var out=new ArrayList<NervFlightProofR52.ObstacleMesh>();for(var raw:j.getAsJsonArray("components")){var c=raw.getAsJsonObject();String name=c.get("part").getAsString();if(name.equals("cage_frame_lower_r44")||name.startsWith("thin_side_rails")||c.has("variant")&&c.get("variant").getAsInt()!=variant)continue;var mesh=gantryMeshes.get(name);if(mesh==null){var a=j.getAsJsonObject("parts").getAsJsonArray(name);double[] xyz=new double[a.size()/2];for(int n=0,k=0;n<a.size();n+=6){xyz[k++]=a.get(n).getAsFloat();xyz[k++]=a.get(n+1).getAsFloat();xyz[k++]=a.get(n+2).getAsFloat();}mesh=new NervRigidMeshSweepR52(xyz);gantryMeshes.put(name,mesh);}Vec3 move=Vec3.ZERO;double open=1-closed;if(c.get("motion").getAsString().equals("translation_only_with_exact_facet_pad")){var a=c.getAsJsonArray("normal");move=new Vec3(a.get(0).getAsDouble(),a.get(1).getAsDouble(),a.get(2).getAsDouble()).scale(2.1*ease(open/.25)).add(c.get("outboard_m").getAsDouble()*ease((open-.23)/.65),0,0);}else if(c.get("motion").getAsString().equals("telescoping_translation")){var a=c.getAsJsonArray("translation_open_local");var interval=c.getAsJsonArray("opening_interval");move=new Vec3(a.get(0).getAsDouble(),a.get(1).getAsDouble(),a.get(2).getAsDouble()).scale(ease((open-interval.get(0).getAsDouble())/(interval.get(1).getAsDouble()-interval.get(0).getAsDouble())));}out.add(new NervFlightProofR52.ObstacleMesh(mesh,frame.compose(new RigidTransform(move,0,0,0,1))));}for(var p:NervLowerFrameR52.pieces(lower))out.add(new NervFlightProofR52.ObstacleMesh(p.mesh(),frame.compose(p.local())));var meshMethod=NervShoulderBearingR52.class.getDeclaredMethod("mesh",String.class,RigidTransform.class,double.class);meshMethod.setAccessible(true);for(var pad:NervShoulderBearingR52.pads())if(pad.variant()==variant){var movement=closed==1?pad.normal().scale(-pad.finalStroke()):pad.normal().scale(2.1).add(pad.side()*5.35,0,0);out.add((NervFlightProofR52.ObstacleMesh)meshMethod.invoke(null,pad.part(),frame.compose(new RigidTransform(movement,0,0,0,1)),1D));for(int i=0;i<2;i++){var anchor=pad.anchors().get(i);var delta=pad.links().get(i).add(movement).subtract(anchor);double length=delta.length();var f=frame.compose(RigidTransform.of(anchor,new Quaternionf().rotationTo(0,0,1,(float)(delta.x/length),(float)(delta.y/length),(float)(delta.z/length))));out.add((NervFlightProofR52.ObstacleMesh)meshMethod.invoke(null,"shoulder_joint",f,1D));out.add((NervFlightProofR52.ObstacleMesh)meshMethod.invoke(null,"shoulder_barrel",f,1D));out.add((NervFlightProofR52.ObstacleMesh)meshMethod.invoke(null,"shoulder_piston_unit",f.compose(new RigidTransform(new Vec3(0,0,2.18),0,0,0,1)),length-2.18));out.add((NervFlightProofR52.ObstacleMesh)meshMethod.invoke(null,"shoulder_joint",f.compose(new RigidTransform(new Vec3(0,0,length),0,0,0,1)),1D));}}return out;}
 private static double ease(double t){t=Math.max(0,Math.min(1,t));return t*t*t*(10+t*(-15+6*t));}
 public static void main(String[] args)throws Exception{load(Path.of(args[2]));var units=JsonParser.parseString(Files.readString(Path.of(args[0]))).getAsJsonObject().getAsJsonArray("units");var gates=JsonParser.parseString(Files.readString(Path.of(args[1]))).getAsJsonObject().getAsJsonArray("gates");var rows=new ArrayList<Object>();
  for(var raw:units){var u=raw.getAsJsonObject();int v=u.get("variant").getAsInt();var bay=frame(u.getAsJsonObject("bay_frame"));JsonObject gate=null;for(var g:gates)if(g.getAsJsonObject().get("resource").getAsString().equals("bay_"+v+"_service_pit"))gate=g.getAsJsonObject();if(gate==null)throw new AssertionError("Root service gate required");var h=gate.getAsJsonObject("silo_hatch");var hs=new SiloHatchMechanism.Spec(h.get("opening_x").getAsFloat(),h.get("opening_z").getAsFloat(),h.get("open_ticks").getAsInt(),h.get("close_ticks").getAsInt(),h.get("separation_sign").getAsInt(),h.get("beam_bottom").getAsDouble(),true);var gf=frame(gate.getAsJsonObject("frame"));var hg=NervSiloHatchGeometryR52.loadClasspath(hs);double top=u.getAsJsonArray("carrier_support").get(0).getAsJsonObject().getAsJsonArray("position").get(1).getAsDouble()+1;var spec=new NervServiceFootDriveR52.Spec(v,bay,gf,hs.firstTier(),top);var hits=new LinkedHashSet<String>();var gatesIdle=new LinkedHashSet<String>();var gatesActive=new LinkedHashSet<String>();var fixedSource=new LinkedHashSet<String>();var actualGantryHits=new LinkedHashSet<String>();var clearCells=new TreeSet<String>();var posts=new ArrayList<Object>();var reservations=new ArrayList<double[]>();AABB bounds=null;
   for(int k=0;k<=128;k++){float rise=k/128F;var sample=new NervLegacyCarrierRideR52.LeadSampleR52(spec,1,rise,0);var lead=NervLegacyCarrierRideR52.actualLeadPartsR52(sample);var original=NervOriginalCarrierR52.partsAtR52(v,bay,rise,new Vec3(0,1,0),NervLegacyCarrierRideR52.rest());var drive=NervServiceFootDriveR52.pieces(spec,1);var actualGantry=gantry(v,frame(u.getAsJsonObject("gantry").getAsJsonObject("frame")),1,1);
    for(var p:lead){var pb=p.mesh().envelope(p.frame(),p.frame());bounds=bounds==null?pb:bounds.minmax(pb);if(k==0&&p.role().equals("lead_fixed_support")){posts.add(Map.of("part",p.name(),"bounds",new double[]{pb.minX,pb.minY,pb.minZ,pb.maxX,pb.maxY,pb.maxZ},"foot_half_xz",new double[]{.12,.12}));for(double low=pb.minY;low<pb.maxY;low+=16)reservations.add(new double[]{pb.minX,low,pb.minZ,pb.maxX,Math.min(pb.maxY,low+16),pb.maxZ});}for(int x=(int)Math.floor(pb.minX+1e-7);x<=Math.floor(pb.maxX-1e-7);x++)for(int y=(int)Math.floor(pb.minY+1e-7);y<=Math.floor(pb.maxY-1e-7);y++)for(int z=(int)Math.floor(pb.minZ+1e-7);z<=Math.floor(pb.maxZ-1e-7);z++)if(p.mesh().intersects(p.frame(),p.frame(),new AABB(x,y,z,x+1,y+1,z+1),.003))clearCells.add(x+"/"+y+"/"+z);if(hg.bounds(1).move(gf.translation()).intersects(pb)&&hg.intersects(gf,1,1,p.mesh(),p.frame(),p.frame()))gatesActive.add(p.name()+" @ "+k);for(var q:original){var mesh=candidates.getOrDefault(q.name(),q.mesh());if(!mesh.envelope(q.frame(),q.frame()).intersects(pb))continue;if(p.role().equals("service_lead_head")&&NervLegacyCarrierRideR52.namedHeadContactR52(q,q,p))continue;if(p.mesh().intersects(p.frame(),p.frame(),mesh,q.frame(),q.frame(),.003,Vec3.ZERO))hits.add(p.name()+" vs "+q.name()+" @ "+k);}for(int gi=0;gi<actualGantry.size();gi++){var q=actualGantry.get(gi);if(q.mesh().envelope(q.frame(),q.frame()).intersects(pb)&&p.mesh().intersects(p.frame(),p.frame(),q.mesh(),q.frame(),q.frame(),.003,Vec3.ZERO))actualGantryHits.add(p.name()+" vs actualGantry/"+gi+" @ "+k);}for(var q:drive)if(q.mesh().envelope(q.frame(),q.frame()).intersects(pb)&&p.mesh().intersects(p.frame(),p.frame(),q.mesh(),q.frame(),q.frame(),.003,Vec3.ZERO))fixedSource.add(p.name()+" vs "+q.name()+" @ "+k);}
    if(k==0)for(var p:lead)if(hg.intersects(gf,0,1,p.mesh(),p.frame(),p.frame()))gatesIdle.add(p.name());
   }
   rows.add(Map.ofEntries(Map.entry("variant",v),Map.entry("carrier_full64_hits",hits),Map.entry("service33_open_hits",gatesActive),Map.entry("service33_full_curve_idle_hits",gatesIdle),Map.entry("actual_service_T_hits",fixedSource),Map.entry("source_bounds",new double[]{bounds.minX,bounds.minY,bounds.minZ,bounds.maxX,bounds.maxY,bounds.maxZ}),Map.entry("required_clear_cell_candidates",clearCells),Map.entry("post_net_bounds",posts),Map.entry("reservation_boxes",reservations),Map.entry("feet",NervLegacyCarrierRideR52.leadFoundationPointsR52(new NervLegacyCarrierRideR52.LeadSampleR52(spec,0,0,0))),Map.entry("actual_gantry_structure_hits",actualGantryHits)));System.out.println("v="+v+" carrier="+hits.size()+" gate_open="+gatesActive.size()+" idle="+gatesIdle.size()+" T="+fixedSource.size()+" gantry="+actualGantryHits.size());Files.writeString(Path.of(args[3]),new GsonBuilder().setPrettyPrinting().create().toJson(Map.of("world_written",false,"native_admission",false,"source",NervLegacyCarrierRideR52.class.getField("SOURCE_SHA").get(null),"actual_source_bytes_sha256",java.util.HexFormat.of().formatHex(java.security.MessageDigest.getInstance("SHA-256").digest(EvaOriginalMeshGeometryR52.classpathBytes(NervLegacyCarrierRideR52.SOURCE))),"rows",rows)));
  }
 }
