package com.projectseele.world;
import com.google.gson.*;
import java.nio.file.*;
import java.util.*;
import net.minecraft.world.phys.*;
/** All distinct measured mechanical conflicts, preserving real source frames. */
public final class TurnAllMechanicalIntersectionsR52 {
 private static Map<String,Object> frame(RigidTransform f){return Map.of("position",new double[]{f.translation().x,f.translation().y,f.translation().z},"quaternion",new double[]{f.qx(),f.qy(),f.qz(),f.qw()});}
 private static double[] bounds(AABB b){return b==null?null:new double[]{b.minX,b.minY,b.minZ,b.maxX,b.maxY,b.maxZ};}
 public static void main(String[] args)throws Exception{
  var d=NervRouteDirectoryR52.parse(JsonParser.parseString(Files.readString(Path.of(args[0]))).getAsJsonObject());var report=JsonParser.parseString(Files.readString(Path.of(args[1]))).getAsJsonObject();var rows=new ArrayList<Object>();var fixedLookup=new HashMap<String,NervServiceFootDriveR52.Piece>();for(var p:NervFixedFootGuidesR52.pieces())fixedLookup.put(p.name(),p);
  for(var raw:report.getAsJsonArray("candidate_stations")){var station=raw.getAsJsonObject();int v=station.get("variant").getAsInt();String leg=station.get("leg").getAsString();var in=point(station.getAsJsonArray("from_up"));var out=point(station.getAsJsonArray("to_up"));var spec=new NervFootGuideDockR52.Spec(d,d.leg(leg),NervOriginalCarrierR52.nominalBinding(v),in,out,station.get("foundation_top").getAsDouble());for(var hit:station.getAsJsonArray("whole_eva_carrier_vs_fixed_and_1m_cubes")){String value=hit.getAsString();int at=value.lastIndexOf(" @ "),split=value.indexOf(" vs ");String moving=value.substring(0,split),fixed=value.substring(split+4,at);int sample=Integer.parseInt(value.substring(at+3));var pose=new NervFootGuideDockR52.Sample(NervFootGuideDockR52.Stage.TURN,sample/16D);var root=NervFootGuideDockR52.root(spec,pose);var part=NervOriginalCarrierR52.partsAtR52(v,root,1,NervFootGuideDockR52.guideUp(spec,pose)).stream().filter(p->p.name().equals(moving)).findFirst().orElseThrow();NervRigidMeshSweepR52 obstacle;RigidTransform obstacleFrame;
    if(fixed.startsWith("future_1m_cube/")){var regex=java.util.regex.Pattern.compile("x=(-?\\d+), y=(-?\\d+), z=(-?\\d+)").matcher(fixed);if(!regex.find())throw new IllegalStateException("Actual native cube coordinates required: "+fixed);var centre=new Vec3(Integer.parseInt(regex.group(1))+.5,Integer.parseInt(regex.group(2))+.5,Integer.parseInt(regex.group(3))+.5);var cube=NervServiceFootDriveR52.box(fixed,new AABB(-.5,-.5,-.5,.5,.5,.5),new RigidTransform(centre,0,0,0,1),false);obstacle=cube.mesh();obstacleFrame=cube.frame();}else{var p=fixedLookup.get(fixed);if(p==null)throw new IllegalStateException("Actual fixed source changed: "+fixed);obstacle=p.mesh();obstacleFrame=p.frame();}
    var xyz=part.mesh().triangles();var indices=new ArrayList<Integer>();var local=new ArrayList<Double>();AABB area=null;for(int n=0;n<xyz.length;n+=9){var tri=Arrays.copyOfRange(xyz,n,n+9);var mesh=new NervRigidMeshSweepR52(tri);if(!mesh.envelope(part.frame(),part.frame()).intersects(obstacle.envelope(obstacleFrame,obstacleFrame)))continue;if(!mesh.intersects(part.frame(),part.frame(),obstacle,obstacleFrame,obstacleFrame,.003,Vec3.ZERO))continue;indices.add(n/9);for(int k=0;k<9;k+=3){var p=new Vec3(tri[k],tri[k+1],tri[k+2]);local.add(p.x);local.add(p.y);local.add(p.z);area=area==null?new AABB(p,p):area.minmax(new AABB(p,p));}}
    var row=new LinkedHashMap<String,Object>();row.put("leg",leg);row.put("variant",v);row.put("sample",sample);row.put("moving",moving);row.put("fixed",fixed);row.put("moving_frame",frame(part.frame()));row.put("fixed_frame",frame(obstacleFrame));row.put("source_indices",indices);row.put("source_local_xyz",local);row.put("source_local_bounds",bounds(area));row.put("fixed_local_xyz",obstacle.triangles());rows.add(row);
  }System.out.println(leg+" cumulative="+rows.size());Files.writeString(Path.of(args[2]),new GsonBuilder().setPrettyPrinting().create().toJson(Map.of("world_written",false,"source",NervFixedFootGuidesR52.SOURCE_SHA,"rows",rows)));}
 }
 private static Vec3 point(JsonArray a){return new Vec3(a.get(0).getAsDouble(),a.get(1).getAsDouble(),a.get(2).getAsDouble());}
}
