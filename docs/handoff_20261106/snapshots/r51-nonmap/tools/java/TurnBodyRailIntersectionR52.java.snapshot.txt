package com.projectseele.world;
import com.google.gson.*;
import com.projectseele.entity.*;
import java.nio.file.*;
import java.util.*;
import net.minecraft.world.phys.*;
/** Exact paired original source evidence for the owner's finite local repair. */
public final class TurnBodyRailIntersectionR52 {
 private static Map<String,Object> frame(RigidTransform f){return Map.of("position",new double[]{f.translation().x,f.translation().y,f.translation().z},"quaternion",new double[]{f.qx(),f.qy(),f.qz(),f.qw()});}
 public static void main(String[] args)throws Exception{
  var d=NervRouteDirectoryR52.parse(JsonParser.parseString(Files.readString(Path.of(args[0]))).getAsJsonObject());var l=d.leg(args[1]);int variant=Integer.parseInt(args[2]);double t=Double.parseDouble(args[3]);var b=NervOriginalCarrierR52.nominalBinding(variant);Vec3 from=null,to=null;for(var route:d.routes().values())if(route.variant()==variant)for(String id:route.legs()){var leg=d.leg(id);if(leg.motion()!=NervRouteDirectoryR52.Motion.VERTICAL&&leg.motion()!=NervRouteDirectoryR52.Motion.LAUNCH_INCLINE)continue;var up=d.node(leg.to()).footFrame().translation().subtract(d.node(leg.from()).footFrame().translation()).normalize();if(up.y<0)up=up.scale(-1);if(leg.to().equals(l.from()))from=up;if(leg.from().equals(l.to()))to=up;}
  var spec=new NervFootGuideDockR52.Spec(d,l,b,from,to,-570);var at=new NervFootGuideDockR52.Sample(NervFootGuideDockR52.Stage.TURN,t);var root=NervFootGuideDockR52.root(spec,at);var carrier=NervOriginalCarrierR52.partsAtR52(variant,root,1,NervFootGuideDockR52.guideUp(spec,at)).stream().filter(p->p.name().equals(args[4])).findFirst().orElseThrow();var fixed=NervFixedFootGuidesR52.pieces().stream().filter(p->p.name().equals(args[5])).findFirst().orElseThrow();var xyz=carrier.mesh().triangles();var rows=new ArrayList<Object>();var localTriangles=new ArrayList<Double>();var worldTriangles=new ArrayList<Double>();AABB local=null,world=null;
  for(int n=0;n<xyz.length;n+=9){var tri=Arrays.copyOfRange(xyz,n,n+9);var mesh=new NervRigidMeshSweepR52(tri);if(!mesh.envelope(carrier.frame(),carrier.frame()).intersects(fixed.mesh().envelope(fixed.frame(),fixed.frame())))continue;if(!mesh.intersects(carrier.frame(),carrier.frame(),fixed.mesh(),fixed.frame(),fixed.frame(),.003,Vec3.ZERO))continue;rows.add(n/9);for(int k=0;k<9;k+=3){var p=new Vec3(tri[k],tri[k+1],tri[k+2]);var q=carrier.frame().transformPoint(p);for(double v:new double[]{p.x,p.y,p.z})localTriangles.add(v);for(double v:new double[]{q.x,q.y,q.z})worldTriangles.add(v);local=local==null?new AABB(p,p):local.minmax(new AABB(p,p));world=world==null?new AABB(q,q):world.minmax(new AABB(q,q));}}
  var result=new LinkedHashMap<String,Object>();result.put("world_written",false);result.put("leg",args[1]);result.put("turn_fraction",t);result.put("carrier_part",carrier.name());result.put("fixed_part",fixed.name());result.put("carrier_frame",frame(carrier.frame()));result.put("fixed_frame",frame(fixed.frame()));result.put("hit_indices",rows);result.put("hit_local_xyz",localTriangles);result.put("hit_world_xyz",worldTriangles);result.put("fixed_local_xyz",fixed.mesh().triangles());result.put("hit_local_bounds",local==null?null:new double[]{local.minX,local.minY,local.minZ,local.maxX,local.maxY,local.maxZ});result.put("hit_world_bounds",world==null?null:new double[]{world.minX,world.minY,world.minZ,world.maxX,world.maxY,world.maxZ});Files.writeString(Path.of(args[6]),new GsonBuilder().setPrettyPrinting().create().toJson(result));System.out.println("source triangles="+rows.size()+" bounds="+local);
 }
}
