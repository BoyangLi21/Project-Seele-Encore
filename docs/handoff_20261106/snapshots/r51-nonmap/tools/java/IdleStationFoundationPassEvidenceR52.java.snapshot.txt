/** Read-only paired source sweep through an idle station, including the
 * preceding/following vertical paths. No actor/world installation occurs. */
public final class IdleStationFoundationPassEvidenceR52 {
 private record Payload(String name,NervRigidMeshSweepR52 mesh){}
 private static List<Payload> payloads(JsonObject goals,int v){for(var raw:goals.getAsJsonArray("variants")){var row=raw.getAsJsonObject();if(row.get("variant").getAsInt()!=v)continue;var out=new ArrayList<Payload>();for(var rawPart:row.getAsJsonArray("parts")){var p=rawPart.getAsJsonObject();var a=p.getAsJsonArray("triangles");double[] xyz=new double[a.size()];for(int i=0;i<a.size();i++)xyz[i]=a.get(i).getAsDouble();out.add(new Payload("EVA/"+p.get("bone").getAsString(),new NervRigidMeshSweepR52(xyz)));}return List.copyOf(out);}throw new AssertionError("Actual goal source variant missing");}
 private static double[] box(AABB b){return new double[]{b.minX,b.minY,b.minZ,b.maxX,b.maxY,b.maxZ};}
 public static void main(String[] args)throws Exception {
  var d=NervRouteDirectoryR52.parse(JsonParser.parseString(Files.readString(Path.of(args[0]))).getAsJsonObject());JsonObject goals;try(var stream=new java.util.zip.GZIPInputStream(Files.newInputStream(Path.of(args[1])))){goals=JsonParser.parseReader(new java.io.InputStreamReader(stream,java.nio.charset.StandardCharsets.UTF_8)).getAsJsonObject();}
  var stationReport=JsonParser.parseString(Files.readString(Path.of(args[2]))).getAsJsonObject();var rows=new ArrayList<Map<String,Object>>();int totalHits=0;
  for(var raw:stationReport.getAsJsonArray("stations")){
   var declared=raw.getAsJsonObject();String stationLeg=declared.get("leg").getAsString();int variant=NervMechanicalRailBindingR52.variant(d,stationLeg);var binding=NervOriginalCarrierR52.nominalBinding(variant);var spec=NervMechanicalRailBindingR52.stationSpec(d,new NervMechanicalRailBindingR52.Spec(NervMechanicalRailBindingR52.Kind.FOOT_TURN_STATION,stationLeg,NervFootGuideDockR52.SOURCE_SHA,NervFootGuideRigR52.SOURCE_SHA,NervFootGuideStationR52.proposedIdentityR52(d.epoch(),stationLeg),declared.get("foundation_top").getAsDouble()));var view=NervFootGuideDockR52.view(spec);var verticals=new TreeSet<String>();for(var route:d.routes().values())for(String id:route.legs()){var leg=d.leg(id);if(leg.motion()==NervRouteDirectoryR52.Motion.VERTICAL)verticals.add(leg.id());}
   var allBodies=new HashMap<Integer,List<Payload>>();for(int v=0;v<3;v++)allBodies.put(v,payloads(goals,v));var hits=new TreeSet<String>();int intervals=0,broadPairs=0;
   for(double turn:new double[]{0,1}){var pose=new StationPose(turn,1,1,0,0);var fixed=new ArrayList<NervFootGuideDockR52.Piece>();for(var at:NervMechanicalRailBindingR52.stationCells(spec)){var cube=NervServiceFootDriveR52.box("native_foot_cube/"+at,new AABB(at),RigidTransform.identity(),false);fixed.add(new NervFootGuideDockR52.Piece(cube.name(),cube.mesh(),cube.rgb(),cube.frame(),false));}var fixedBounds=new ArrayList<AABB>();AABB stationBounds=null;for(var p:fixed){var b=p.mesh().envelope(p.frame(),p.frame());fixedBounds.add(b);stationBounds=stationBounds==null?b:stationBounds.minmax(b);}
    for(String id:verticals){var leg=d.leg(id);int movingVariant=NervMechanicalRailBindingR52.variant(d,id);var movingBinding=NervOriginalCarrierR52.nominalBinding(movingVariant);var body=allBodies.get(movingVariant);var up=NervFixedFootGuidesR52.up(d,leg);var lineA=d.sample(leg,0,movingBinding).evaRoot();var lineB=d.sample(leg,1,movingBinding).evaRoot();var ca=NervOriginalCarrierR52.partsAtR52(movingVariant,lineA,1,up,NervLegacyCarrierRideR52.rest());var cb=NervOriginalCarrierR52.partsAtR52(movingVariant,lineB,1,up,NervLegacyCarrierRideR52.rest());AABB lineBounds=null;for(int j=0;j<ca.size();j++){var bound=ca.get(j).mesh().envelope(ca.get(j).frame(),cb.get(j).frame());lineBounds=lineBounds==null?bound:lineBounds.minmax(bound);}for(var q:body)lineBounds=lineBounds.minmax(q.mesh().envelope(lineA,lineB));if(!lineBounds.intersects(stationBounds))continue;double length=d.node(leg.from()).footFrame().translation().distanceTo(d.node(leg.to()).footFrame().translation());int divisions=Math.max(1,(int)Math.ceil(length));for(int i=0;i<divisions;i++){double a=i/(double)divisions,b=(i+1)/(double)divisions;var rootA=d.sample(leg,a,movingBinding).evaRoot();var rootB=d.sample(leg,b,movingBinding).evaRoot();var from=NervOriginalCarrierR52.partsAtR52(movingVariant,rootA,1,up,NervLegacyCarrierRideR52.rest());var to=NervOriginalCarrierR52.partsAtR52(movingVariant,rootB,1,up,NervLegacyCarrierRideR52.rest());AABB assembly=null;for(int j=0;j<from.size();j++){var bound=from.get(j).mesh().envelope(from.get(j).frame(),to.get(j).frame());assembly=assembly==null?bound:assembly.minmax(bound);}for(var p:body)assembly=assembly.minmax(p.mesh().envelope(rootA,rootB));if(!assembly.intersects(stationBounds))continue;intervals++;
     for(int j=0;j<fixed.size();j++){var p=fixed.get(j);var fb=fixedBounds.get(j);for(int k=0;k<from.size();k++){var x=from.get(k);var y=to.get(k);if(!x.mesh().envelope(x.frame(),y.frame()).intersects(fb))continue;broadPairs++;if(x.mesh().intersects(x.frame(),y.frame(),p.mesh(),p.frame(),p.frame(),.003,Vec3.ZERO))hits.add("turn="+turn+" path="+id+" interval="+i+" carrier="+x.name()+" station="+p.name());}for(var q:body){if(!q.mesh().envelope(rootA,rootB).intersects(fb))continue;broadPairs++;if(q.mesh().intersects(rootA,rootB,p.mesh(),p.frame(),p.frame(),.003,Vec3.ZERO))hits.add("turn="+turn+" path="+id+" interval="+i+" "+q.name()+" station="+p.name());}}
    }}
   }
   totalHits+=hits.size();rows.add(Map.of("station",stationLeg,"variant",variant,"idle_turns",List.of(0,1),"all_lanes_vertical_paths",verticals,"relevant_source_intervals",intervals,"broad_pairs",broadPairs,"source_hits",hits));System.out.println(stationLeg+" idle vertical source hits="+hits.size());Files.writeString(Path.of(args[3]),new GsonBuilder().setPrettyPrinting().create().toJson(Map.of("world_written",false,"native_accepted",false,"goal_scope","Actual selected goal skin + representative34 original hand joints, complete selected carrier/shoes/guides/4bogies. Actual native serviceHeld and pilot are not claimed.","full_source_sha256",NervOriginalCarrierR52.class.getField("SOURCE_SHA").get(null),"dock_source_sha256",NervFootGuideDockR52.class.getField("SOURCE_SHA").get(null),"fixed_source_sha256",NervFixedFootGuidesR52.SOURCE_SHA,"sweep_interval_maximum_m",1,"total_hits",totalHits,"stations",rows)));
  }
 }
}
